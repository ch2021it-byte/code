// ===================================
// Upload Photo
// ===================================
function uploadPhoto(photoData, fileName) {
  try {
    // ຖ້າເປັນ URL ຢູ່ແລ້ວ (ຮູບທີ່ມີຢູ່) ໃຫ້ສົ່ງຄືນຄ່າກັບໄປເລີຍ
    if (!photoData) {
      return '';
    }
    
    if (photoData.startsWith('https://drive.google.com')) {
      Logger.log('Photo already uploaded, reusing URL: ' + photoData);
      return photoData; // ໃຊ້ URL ເດີມ ບໍ່ອັບໂຫລດຊ້ຳ
    }
    
    if (!photoData.startsWith('data:image')) {
      return '';
    }
    
    const base64Data = photoData.split(',')[1];
    const blob = Utilities.newBlob(Utilities.base64Decode(base64Data), 'image/png', fileName);
    const folder = DriveApp.getFolderById(PHOTO_FOLDER_ID);
    
    // ກວດສອບວ່າມີໄຟລ໌ຊື່ນີ້ຢູ່ແລ້ວຫຼືບໍ່
    const existingFiles = folder.getFilesByName(fileName);
    if (existingFiles.hasNext()) {
      const existingFile = existingFiles.next();
      Logger.log('File exists, reusing URL: ' + existingFile.getUrl());
      return existingFile.getUrl(); // ໃຊ້ URL ເດີມ
    }
    
    // ອັບໂຫລດໃໝ່
    const file = folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    
    Logger.log('New file uploaded: ' + file.getUrl());
    return file.getUrl();
    
  } catch (error) {
    Logger.log('Error uploading photo: ' + error.message);
    return photoData;
  }
}

function getLoginData() {
  try {
    Logger.log('=== Starting getLoginData ===');
    const startTime = new Date().getTime();
    
    const cache = CacheService.getScriptCache();
    const cacheKey = 'loginData_v5';
    const cachedData = cache.get(cacheKey);
    
    if (cachedData) {
      Logger.log('✅ Using cached login data (instant load)');
      const endTime = new Date().getTime();
      Logger.log('Time taken: ' + (endTime - startTime) + ' ms');
      return JSON.parse(cachedData);
    }
    
    Logger.log('📥 Fetching fresh data from sheet...');
    Logger.log('Using LOGIN Sheet ID: ' + SHEET_IDS.LOGIN);
    
    const ss = SpreadsheetApp.openById(SHEET_IDS.LOGIN);
    
    // ===== ✅ ພາກທີ 1: ດຶງລາຍຊື່ ແຂວງ/ເມືອງ/ບ້ານ ຈາກຕາຕະລາງ "village" (ຕົວຈິງ 48 ບ້ານ) =====
    const villageSheet = ss.getSheetByName(SHEET_NAMES.VILLAGE);
    if (!villageSheet) {
      throw new Error('Sheet not found: ' + SHEET_NAMES.VILLAGE);
    }
    
    const villageLastRow = villageSheet.getLastRow();
    const provinces = new Set();
    const districts = {};
    const villages = {};
    
    if (villageLastRow >= 2) {
      // ອ່ານຖັນ B-D (village, district, province) ຕາມທີ່ validateLogin ໃຊ້
      const villageData = villageSheet.getRange(2, 2, villageLastRow - 1, 3).getValues(); // B,C,D
      
      villageData.forEach(function(row) {
        const village = row[0] ? String(row[0]).trim() : '';
        const district = row[1] ? String(row[1]).trim() : '';
        const province = row[2] ? String(row[2]).trim() : '';
        
        if (!village || !district || !province) {
          return;
        }
        
        provinces.add(province);
        
        if (!districts[province]) {
          districts[province] = new Set();
        }
        districts[province].add(district);
        
        const key = province + '|' + district;
        if (!villages[key]) {
          villages[key] = new Set();
        }
        villages[key].add(village);
      });
    }
    
    Logger.log('✅ Villages loaded from master "village" sheet: ' + villageLastRow + ' rows');
    
    // ===== ພາກທີ 2: ດຶງຂໍ້ມູນຜູ້ໃຊ້ (ອສບ) ຈາກຕາຕະລາງ "ປະຫວັດ ອສບ" ຄືເດີມ ສຳລັບ login step ຕໍ່ໄປ =====
    const sheet = ss.getSheetByName(SHEET_NAMES.USER);
    
    if (!sheet) {
      throw new Error('Sheet not found: ' + SHEET_NAMES.USER);
    }
    
    const lastRow = sheet.getLastRow();
    Logger.log('Last row: ' + lastRow);
    
    const users = [];
    
    if (lastRow >= 2) {
      const data = sheet.getRange(2, 1, lastRow - 1, 20).getValues();
      Logger.log('User data retrieved: ' + data.length + ' rows');
      
      data.forEach(function(row) {
        const code = row[0] ? String(row[0]).trim() : '';
        const name = row[1] ? String(row[1]).trim() : '';
        const phone = row[8] ? String(row[8]).trim() : '';
        const village = row[4] ? String(row[4]).trim() : '';
        const district = row[18] ? String(row[18]).trim() : '';
        const province = row[19] ? String(row[19]).trim() : '';
        
        if (!code || !name || !village || !district || !province) {
          return;
        }
        
        users.push({
          code: code,
          name: name,
          phone: phone,
          village: village,
          district: district,
          province: province
        });
      });
    }
    
    const result = {
      provinces: Array.from(provinces).sort(),
      districts: {},
      villages: {},
      users: users
    };
    
    Object.keys(districts).forEach(function(province) {
      result.districts[province] = Array.from(districts[province]).sort();
    });
    
    Object.keys(villages).forEach(function(key) {
      result.villages[key] = Array.from(villages[key]).sort();
    });
    
    const endTime = new Date().getTime();
    Logger.log('✅ Login data loaded successfully');
    Logger.log('Provinces: ' + result.provinces.length);
    Logger.log('Users: ' + result.users.length);
    Logger.log('Time taken: ' + (endTime - startTime) + ' ms');
    
    try {
      const jsonString = JSON.stringify(result);
      const maxSize = 100000;
      
      if (jsonString.length < maxSize) {
        cache.put(cacheKey, jsonString, 600);
        Logger.log('✅ Login data cached for 10 minutes');
      } else {
        Logger.log('⚠️ Data too large for cache: ' + jsonString.length + ' bytes');
      }
    } catch (e) {
      Logger.log('⚠️ Cache error: ' + e.message);
    }
    
    return result;
    
  } catch (error) {
    Logger.log('❌ ERROR in getLoginData: ' + error.message);
    Logger.log('Stack: ' + error.stack);
    throw new Error('Error loading login data: ' + error.message);
  }
}

// ===================================
// Validate Login & Get Village Sheet ID
// ===================================
function validateLogin(loginData) {
  try {
    Logger.log('=== Validating Login ===');
    Logger.log('Input: ' + JSON.stringify(loginData));
    
    const ss = SpreadsheetApp.openById(SHEET_IDS.LOGIN);
    Logger.log('LOGIN sheet opened');
    
    const villageSheet = ss.getSheetByName(SHEET_NAMES.VILLAGE);
    Logger.log('Village sheet: ' + (villageSheet ? 'found' : 'NOT FOUND'));
    
    if (!villageSheet) {
      return {
        success: false,
        message: 'Village sheet not found in LOGIN'
      };
    }
    
    const villageData = villageSheet.getDataRange().getValues();
    Logger.log('Village data rows: ' + villageData.length);
    
    const inputVillage = String(loginData.village).trim();
    const inputDistrict = String(loginData.district).trim();
    const inputProvince = String(loginData.province).trim();
    
    Logger.log('Looking for: V=' + inputVillage + ', D=' + inputDistrict + ', P=' + inputProvince);
    
    // Search in village sheet
    // Column B = village (index 1)
    // Column C = district (index 2)
    // Column D = province (index 3)
    // Column E = Sheet ID (index 4)
    
    for (let i = 1; i < villageData.length; i++) {
      const row = villageData[i];
      
      const rowVillage = row[1] ? String(row[1]).trim() : '';
      const rowDistrict = row[2] ? String(row[2]).trim() : '';
      const rowProvince = row[3] ? String(row[3]).trim() : '';
      const sheetId = row[4] ? String(row[4]).trim() : '';
      
      Logger.log('Row ' + i + ': V=' + rowVillage + ', D=' + rowDistrict + ', P=' + rowProvince + ', ID=' + sheetId);
      
      if (rowVillage === inputVillage && rowDistrict === inputDistrict && rowProvince === inputProvince) {
        Logger.log('✅ MATCH FOUND! Sheet ID: ' + sheetId);
        
        return {
          success: true,
          villageSheetId: sheetId,
          village: inputVillage,
          district: inputDistrict,
          province: inputProvince
        };
      }
    }
    
    Logger.log('❌ No match found');
    return {
      success: false,
      message: 'Village not found in village sheet'
    };
    
  } catch (error) {
    Logger.log('ERROR in validateLogin: ' + error.message);
    Logger.log('Stack: ' + error.stack);
    return {
      success: false,
      message: 'Error: ' + error.message
    };
  }
}

// ===================================
// Get Options Data
// ===================================
function getOptionsData() {
  try {
    const ss = SpreadsheetApp.openById(SHEET_IDS.OPTIONS);
    const sheet = ss.getSheetByName(SHEET_NAMES.OPTIONS);
    const data = sheet.getDataRange().getValues();
    
    const options = {};
    const headers = data[0];
    
    for (let col = 0; col < headers.length; col++) {
      const columnData = [];
      for (let row = 1; row < data.length; row++) {
        if (data[row][col]) {
          columnData.push(data[row][col]);
        }
      }
      options[String.fromCharCode(65 + col)] = columnData;
    }
    
    return options;
  } catch (error) {
    throw new Error('Error loading options: ' + error.message);
  }
}
// ===================================
// Helper: Format Birth Date (ແກ້ໄຂບັນຫາ Google Sheets ປ່ຽນຂໍ້ຄວາມວັນທີເປັນ Date ອັດຕະໂນມັດ)
// ===================================
function formatBirthDate(value) {
  if (!value) return '';
  
  if (value instanceof Date) {
    const day = value.getDate();
    const month = value.getMonth() + 1;
    const year = value.getFullYear();
    return day + '/' + month + '/' + year;
  }
  
  return String(value).trim();
}

      success: true,
      message: 'Deleted successfully'
    };

    };
  }

function generateCode(length) {
  let code = '';
  for (let i = 0; i < length; i++) {
    code += Math.floor(Math.random() * 10);
  }
  return code;
}

function deletePhotoFromDrive(photoUrl) {
  try {
    if (!photoUrl || !photoUrl.includes('drive.google.com')) {
      return;
    }
    
    const match = photoUrl.match(/[-\w]{25,}/);
    if (!match) {
      return;
    }
    
    const fileId = match[0];
    const file = DriveApp.getFileById(fileId);
    
    if (file) {
      file.setTrashed(true);
    }
    
  } catch (error) {
    Logger.log('Error deleting photo: ' + error.message);
  }
}

function searchHouseholds(searchQuery, searchField, villageSheetId) {
  try {
    if (!villageSheetId) {
      throw new Error('No village sheet ID');
    }
    
    const ss = SpreadsheetApp.openById(villageSheetId);
    const householdSheet = ss.getSheetByName(SHEET_NAMES.HOUSEHOLD);
    const memberSheet = ss.getSheetByName(SHEET_NAMES.MEMBER);
    
    if (!householdSheet) {
      return [];
    }
    
    const householdData = householdSheet.getDataRange().getValues();
    const memberData = memberSheet ? memberSheet.getDataRange().getValues() : [];
    const results = [];
    const addedHouseholds = new Set(); // ປ້ອງກັນຊ້ຳ
    
    const searchLower = searchQuery.toLowerCase();
    
    // ຄົ້ນຫາຈາກຫົວໜ້າຄອບຄົວ
    for (let i = 1; i < householdData.length; i++) {
      const row = householdData[i];
      let match = false;
      
      if (searchField === 'name') {
        match = row[1] && String(row[1]).toLowerCase().includes(searchLower);
      } else if (searchField === 'phone') {
        match = row[3] && String(row[3]).includes(searchQuery);
      } else if (searchField === 'householdCode') {
        match = row[0] && String(row[0]).includes(searchQuery);
      } else if (searchField === 'houseNumber') {
        match = row[4] && String(row[4]).toLowerCase().includes(searchLower);
      } else {
        // ຄົ້ນຫາທຸກຟິລດ໌
        match = (row[0] && String(row[0]).toLowerCase().includes(searchLower)) ||
                (row[1] && String(row[1]).toLowerCase().includes(searchLower)) ||
                (row[3] && String(row[3]).includes(searchQuery)) ||
                (row[4] && String(row[4]).toLowerCase().includes(searchLower));
      }
      
      if (match) {
        const householdCode = String(row[0]);
        if (!addedHouseholds.has(householdCode)) {
          results.push({
            householdCode: householdCode,
            name: String(row[1] || ''),
            photo: String(row[2] || ''),

        }
      }
    }
    
    // ຄົ້ນຫາຈາກສະມາຊິກ
    if (memberData.length > 1) {
      for (let i = 1; i < memberData.length; i++) {
        const row = memberData[i];
        const householdCode = String(row[0]);
        
        if (addedHouseholds.has(householdCode)) {
          continue; // ຖ້າພົບແລ້ວຂ້າມ
        }
        
        let match = false;
        
        if (searchField === 'name') {
          match = row[2] && String(row[2]).toLowerCase().includes(searchLower);
        } else if (searchField === 'phone') {
          match = row[13] && String(row[13]).includes(searchQuery);
        } else {
          // ຄົ້ນຫາທຸກຟິລດ໌ຂອງສະມາຊິກ
          match = (row[2] && String(row[2]).toLowerCase().includes(searchLower)) ||
                  (row[13] && String(row[13]).includes(searchQuery));
        }
        
        if (match) {
          // ດຶງຂໍ້ມູນຄອບຄົວຂອງສະມາຊິກຄົນນີ້
          const householdRow = householdData.find(h => String(h[0]) === householdCode);

              householdCode: householdCode,
              name: String(householdRow[1] || ''),

              phone: String(householdRow[3] || ''),

              registryNumber: String(householdRow[9] || ''),
              familyMemberCount: String(householdRow[10] || ''),
              pageCount: String(householdRow[11] || ''),
              unit: String(householdRow[12] || ''),
              rentalLocation: String(householdRow[13] || ''),
              monthlyIncome: String(householdRow[14] || ''),
              housePosition: String(householdRow[15] || ''),
              village: String(householdRow[16] || ''),
              district: String(householdRow[17] || ''),
              province: String(householdRow[18] || ''),
              userCode: String(householdRow[19] || ''),

// ===================================
// Parse Rental Periods (Z-AA)
// ===================================
function parseRentalPeriods(renewalData, endDateData) {
  const periods = [];
  
  if (!renewalData || !endDateData) {
    return periods;
  }
  
  // Parse format: "ຄັ້ງທີ 1: 01/01/2025 (30 ວັນ) | ຄັ້ງທີ 2: 01/02/2025 (60 ວັນ)"
  const renewalStr = String(renewalData);
  const endDateStr = String(endDateData);
  
  const renewalParts = renewalStr.split('|');
  const endDateParts = endDateStr.split('|');
  
  for (let i = 0; i < renewalParts.length; i++) {
    const renewal = renewalParts[i].trim();
    const endDate = endDateParts[i] ? endDateParts[i].trim() : '';
    
    // Extract date and days
    const dateMatch = renewal.match(/(\d{2}\/\d{2}\/\d{4})/);
    const daysMatch = renewal.match(/\((\d+)\s*[^\)]*\)/);
    
    if (dateMatch) {
      periods.push({
        periodNumber: i + 1,
        renewalDate: dateMatch[1],
        days: daysMatch ? parseInt(daysMatch[1]) : 0,
        endDate: endDate
      });
    }
  }
  
  return periods;
}

// Search Mobile Population
// ===================================
function searchMobile(searchQuery, villageSheetId) {
  try {
    if (!villageSheetId) {
      throw new Error('No village sheet ID');
    }
    
    const ss = SpreadsheetApp.openById(villageSheetId);
    const mobileSheet = ss.getSheetByName(SHEET_NAMES_MOBILE.MOBILE);
    
    if (!mobileSheet) {
      return [];
    }
    
    const data = mobileSheet.getDataRange().getValues();
    const results = [];
    const searchLower = searchQuery.toLowerCase();
    
    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      
      // Search in: name, phone, room number, owner name
      const match = (row[9] && String(row[9]).toLowerCase().includes(searchLower)) ||  // J - name
                    (row[18] && String(row[18]).includes(searchQuery)) ||              // S - phone
                    (row[7] && String(row[7]).toLowerCase().includes(searchLower)) ||  // H - room number
                    (row[27] && String(row[27]).toLowerCase().includes(searchLower));  // AB - owner name
      
      if (match) {
        const rentalPeriods = parseRentalPeriods(row[25], row[26]);
        
        results.push({
          mobileCode: String(row[0] || ''),
          householdCode: String(row[1] || ''),
          province: String(row[2] || ''),
          district: String(row[3] || ''),
          village: String(row[4] || ''),
          unit: String(row[5] || ''),
          position: String(row[6] || ''),
          roomNumber: String(row[7] || ''),
          roomCount: String(row[8] || ''),
          name: String(row[9] || ''),
          birthDate: String(row[10] || ''),
          gender: String(row[11] || ''),
          nationality: String(row[12] || ''),
          ethnicity: String(row[13] || ''),
          religion: String(row[14] || ''),
          education: String(row[15] || ''),
          relationship: String(row[16] || ''),
          maritalStatus: String(row[17] || ''),
          phone: String(row[18] || ''),
          occupation: String(row[19] || ''),
          workplace: String(row[20] || ''),
          purpose: String(row[21] || ''),
          photo: String(row[22] || ''),
          hasPermit: String(row[23] || ''),
          firstArrival: String(row[24] || ''),
          rentalPeriods: rentalPeriods,
          ownerName: String(row[27] || ''),
          rentalType: row[1] ? 'ພັກເຊົາຄົງທີ່' : 'ພັກເຊົາເຄື່ອນທີ່'
        });
      }
    }
    
    Logger.log('Mobile search found: ' + results.length);
    return results;
    
  } catch (error) {
    Logger.log('ERROR in searchMobile: ' + error.message);
    return [];
  }
}

// ===================================
// Helper: Format Rental Periods
// ===================================
function formatRentalPeriods(periods) {
  if (!periods || periods.length === 0) {
    return '';
  }
  
  const parts = periods.map(function(p) {
    return 'ຄັ້ງທີ່ ' + p.periodNumber + ': ' + p.renewalDate + ' (' + p.days + ' ວັນ)';
  });
  
  return parts.join(' | ');
}

// ===================================
// Helper: Format End Dates
// ===================================
function formatEndDates(periods) {
  if (!periods || periods.length === 0) {
    return '';
  }
  
  const parts = periods.map(function(p) {
    return 'ຄັ້ງທີ່ ' + p.periodNumber + ': ' + p.endDate;
  });
  
  return parts.join(' | ');
}

// ===================================
// Calculate End Date
// ===================================
function calculateEndDate(startDate, days) {
  try {
    // Parse dd/mm/yyyy
    const parts = startDate.split('/');
    if (parts.length !== 3) {
      return '';
    }
    
    const day = parseInt(parts[0]);
    const month = parseInt(parts[1]) - 1; // JS months are 0-based
    const year = parseInt(parts[2]);
    
    const date = new Date(year, month, day);
    date.setDate(date.getDate() + parseInt(days));
    
    const endDay = String(date.getDate()).padStart(2, '0');
    const endMonth = String(date.getMonth() + 1).padStart(2, '0');
    const endYear = date.getFullYear();
    
    return endDay + '/' + endMonth + '/' + endYear;
  } catch (error) {
    Logger.log('Error calculating end date: ' + error.message);
    return '';
  }
}

          digitsOnly: digitsOnly
