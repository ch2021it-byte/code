// SAVE RECORD — รองรับ col U (lat=index 20), V (lng=index 21)
// ============================================
function saveRecord(record) {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var ws = ss.getSheetByName(SHEET_DATA);
    var data = ws.getDataRange().getValues();
    
    var rowIndex = -1;
    for(var i = 1; i < data.length; i++) {
      if(data[i][0] == record.osbCode) {
        rowIndex = i + 1;
        break;
      }
    }
    
    var rowData = [
      record.osbCode, record.fullName, record.gender, record.dob, record.phone,
      record.province, record.city, record.village, record.unit, record.disease,
      record.schoolProvince, record.schoolCity, record.schoolName, record.classLevel,
      record.status, record.trainingDate, record.trainingTopics, record.hasCert,
      record.profileImg, record.certImg,
      record.lat || '', record.lng || ''
    ];
    
    if(rowIndex > 0) {
      ws.getRange(rowIndex, 1, 1, 22).setValues([rowData]);
      return {success: true, message: "ອັບເດດຂໍ້ມູນສຳເລັດ", mode: "edit"};
    } else {
      ws.appendRow(rowData);
      return {success: true, message: "ເພີ່ມຂໍ້ມູນສຳເລັດ", mode: "add"};
    }
  } catch(error) {
    return {success: false, message: error.toString()};
  }
}

// GENERATE OSB CODE
// ============================================
function generateOsbCode() {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var ws = ss.getSheetByName(SHEET_DATA);
    var data = ws.getDataRange().getValues();
    
    var maxCode = 0;
    for(var i = 1; i < data.length; i++) {
      var code = String(data[i][0]).replace(/\D/g, '');
      if(code && !isNaN(code)) {
        var num = parseInt(code);
        if(num > maxCode) maxCode = num;
      }
    }
    var newCode = String(maxCode + 1).padStart(5, '0');
    return {success: true, code: newCode};
  } catch(error) {
    return {success: false, message: error.toString()};
  }
}

// GET SLICE DATA
// ============================================
function getSliceData(column) {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var ws = ss.getSheetByName(SHEET_SLICE);
    
    var columnMap = {
      'province': 1, 'city': 2, 'village': 3,
      'schoolProvince': 4, 'schoolCity': 5, 'schoolName': 6,
      'disease': 7, 'trainingTopics': 8
    };
    
    var colIndex = columnMap[column];
    if(!colIndex) return {success: false, message: "Invalid column: " + column};
    
    var values = ws.getRange(1, colIndex, ws.getLastRow(), 1).getValues();
    var uniqueValues = [];
    for(var i = 0; i < values.length; i++) {
      var val = String(values[i][0]).trim();
      if(val && uniqueValues.indexOf(val) === -1) uniqueValues.push(val);
    }
    return {success: true, data: uniqueValues};
  } catch(error) {
    return {success: false, message: error.toString()};
  }
}

// ============================================
// GET ALL SLICE DATA + location data (col I=lat, J=lng)
// ============================================
function getAllSliceData() {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var ws = ss.getSheetByName(SHEET_SLICE);
    var lastRow = ws.getLastRow();
    // คอลัมน์ A-J (1-10): A=province, B=city, C=village, D=schoolProv, E=schoolCity, F=schoolName, G=disease, H=trainingTopics, I=lat, J=lng
    var allData = ws.getRange(1, 1, lastRow, 10).getValues();
    
    var result = {
      province: [], city: [], village: [],
      schoolProvince: [], schoolCity: [], schoolName: [],
      disease: [], trainingTopics: [],
      cityByProvince: {}, villageByCity: {},
      schoolCityByProvince: {}, schoolNameByCity: {},
      // Location maps: key=province/city/village → {lat, lng}
      locationByProvince: {},
      locationByCity: {},
      locationByVillage: {}
    };
    
    for(var i = 0; i < allData.length; i++) {
      var row = allData[i];
      var prov  = String(row[0] || '').trim();
      var city  = String(row[1] || '').trim();
      var vill  = String(row[2] || '').trim();
      var sProv = String(row[3] || '').trim();
      var sCity = String(row[4] || '').trim();
      var sName = String(row[5] || '').trim();
      var dis   = String(row[6] || '').trim();
      var train = String(row[7] || '').trim();
      var lat   = row[8] ? parseFloat(row[8]) : null;
      var lng   = row[9] ? parseFloat(row[9]) : null;
      
      if(prov  && result.province.indexOf(prov) === -1)   result.province.push(prov);
      if(city  && result.city.indexOf(city) === -1)        result.city.push(city);
      if(vill  && result.village.indexOf(vill) === -1)     result.village.push(vill);
      if(sProv && result.schoolProvince.indexOf(sProv) === -1) result.schoolProvince.push(sProv);
      if(sCity && result.schoolCity.indexOf(sCity) === -1) result.schoolCity.push(sCity);
      if(sName && result.schoolName.indexOf(sName) === -1) result.schoolName.push(sName);
      if(dis   && result.disease.indexOf(dis) === -1)      result.disease.push(dis);
      if(train && result.trainingTopics.indexOf(train) === -1) result.trainingTopics.push(train);
      
      if(prov && city) {
        if(!result.cityByProvince[prov]) result.cityByProvince[prov] = [];
        if(result.cityByProvince[prov].indexOf(city) === -1) result.cityByProvince[prov].push(city);
      }
      if(city && vill) {
        if(!result.villageByCity[city]) result.villageByCity[city] = [];
        if(result.villageByCity[city].indexOf(vill) === -1) result.villageByCity[city].push(vill);
      }
      if(sProv && sCity) {
        if(!result.schoolCityByProvince[sProv]) result.schoolCityByProvince[sProv] = [];
        if(result.schoolCityByProvince[sProv].indexOf(sCity) === -1) result.schoolCityByProvince[sProv].push(sCity);
      }
      if(sCity && sName) {
        if(!result.schoolNameByCity[sCity]) result.schoolNameByCity[sCity] = [];
        if(result.schoolNameByCity[sCity].indexOf(sName) === -1) result.schoolNameByCity[sCity].push(sName);
      }
      
      // Store location data at each level
      if(lat && lng) {
        if(vill && !result.locationByVillage[vill]) {
          result.locationByVillage[vill] = {lat: lat, lng: lng, province: prov, city: city};
        }
        if(city && !result.locationByCity[city]) {
          result.locationByCity[city] = {lat: lat, lng: lng, province: prov};
        }
        if(prov && !result.locationByProvince[prov]) {
          result.locationByProvince[prov] = {lat: lat, lng: lng};
        }
      }
    }
    
    return {success: true, data: result};
  } catch(error) {
    return {success: false, message: error.toString()};
  }
}

// ============================================
// GET LOCATION DATA — ดึง lat/lng จาก Slice ตาม province/city/village
// ============================================
function getLocationData(province, city, village) {
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var ws = ss.getSheetByName(SHEET_SLICE);
    var lastRow = ws.getLastRow();
    var allData = ws.getRange(1, 1, lastRow, 10).getValues();
    
    for(var i = 0; i < allData.length; i++) {
      var row = allData[i];
      var rProv = String(row[0] || '').trim();
      var rCity = String(row[1] || '').trim();
      var rVill = String(row[2] || '').trim();
      var lat   = row[8] ? parseFloat(row[8]) : null;
      var lng   = row[9] ? parseFloat(row[9]) : null;
      
      // Match priority: village > city > province
      if(village && rVill === village && lat && lng) {
        return {success: true, lat: lat, lng: lng, level: 'village'};
      }
    }
    // Try city level
    for(var i = 0; i < allData.length; i++) {
      var row = allData[i];
      var rCity = String(row[1] || '').trim();
      var lat   = row[8] ? parseFloat(row[8]) : null;
      var lng   = row[9] ? parseFloat(row[9]) : null;
      if(city && rCity === city && lat && lng) {
        return {success: true, lat: lat, lng: lng, level: 'city'};
      }
    }
    // Try province level
    for(var i = 0; i < allData.length; i++) {
      var row = allData[i];
      var rProv = String(row[0] || '').trim();
      var lat   = row[8] ? parseFloat(row[8]) : null;
      var lng   = row[9] ? parseFloat(row[9]) : null;
      if(province && rProv === province && lat && lng) {
        return {success: true, lat: lat, lng: lng, level: 'province'};
      }
    }
    
    return {success: false, message: 'ບໍ່ພົບຂໍ້ມູນພິກັດ'};
  } catch(error) {
    return {success: false, message: error.toString()};
  }
}

// ============================================
// HELPER FUNCTIONS
// ============================================
function formatDate(date) {
  if(!date) return '';
  if(typeof date === 'string') return date;
  try {
    var d = new Date(date);
    var day = String(d.getDate()).padStart(2, '0');
    var month = String(d.getMonth() + 1).padStart(2, '0');
    var year = d.getFullYear();
    return day + '/' + month + '/' + year;
  } catch(e) {
    return String(date);
  }
}

function deleteOldImage(fileUrl) {
  try {
    if(!fileUrl) return {success: false, message: 'No URL'};
    var raw = String(fileUrl);
    var fileId = '';
    if(raw.indexOf('/d/') > -1) fileId = raw.split('/d/')[1].split('/')[0];
    else if(raw.indexOf('id=') > -1) fileId = raw.split('id=')[1].split('&')[0];
    if(!fileId) return {success: false, message: 'ບໍ່ພົບ File ID'};
    DriveApp.getFileById(fileId).setTrashed(true);
    return {success: true, message: 'ລຶບຮູບເກົ່າສຳເລັດ'};
  } catch(error) {
    return {success: false, message: error.toString()};
  }
}

      var blob = Utilities.newBlob(
        Utilities.base64Decode(data.profileImgData.split(',')[1]),
        'image/jpeg',

      );
      var file = folder.createFile(blob);
      imgUrl = 'https://drive.google.com/file/d/' + file.getId() + '/view?usp=sharing';
    }

    // เพิ่มแถวใหม่ — role=ຜູ້ໃຊ້, status=ລໍຖ້າການອະນຸມັດ
    ws.appendRow([
      data.username,
      hashed,
      data.fullName,
      data.province,
      data.city,
      imgUrl,
      'ຜູ້ໃຊ້',
      'ລໍຖ້າການອະນຸມັດ'
    ]);

    return { success: true, message: 'ລົງທະບຽນສຳເລັດ ກະລຸນາລໍຖ້າການອະນຸມັດຈາກ Admin' };
  } catch(e) {
    return { success: false, message: e.toString() };
  }
}
function loginUser(username, password) {

    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);

    Logger.log('username input: ' + username);
    Logger.log('password input: ' + password);

    for (var i = 1; i < all.length; i++) {
      var row = all[i];
      Logger.log('row[0]: ' + row[0] + ' | row[1]: ' + row[1]);

      if (String(row[0]).toLowerCase() === String(username).toLowerCase()) {
        if (String(row[1]).trim() !== String(password).trim()) {
          return { success: false, message: 'ລະຫັດຜ່ານບໍ່ຖືກຕ້ອງ' };
        }
        if (String(row[7]).trim() !== 'ອະນຸມັດ') {
          return { success: false, message: 'ບັນຊີຂອງທ່ານຍັງລໍຖ້າການອະນຸມັດຈາກ Admin' };
        }
        return {
          success: true,
          user: {
            username: row[0],
            fullName: row[2],
            province: row[3],
            city: row[4],
            profileImg: row[5],
            role: row[6],
            isAdmin: String(row[6]).trim() === 'ຜູ້ດູແລລະບົບ'
          }
        };
      }
    }

    return { success: false, message: e.toString() };
  }
}

var DB_NAME='OSB_DB', DB_VERSION=1, db=null, isOnline=navigator.onLine;
function initDB(){return new Promise(function(res,rej){var r=indexedDB.open(DB_NAME,DB_VERSION);r.onupgradeneeded=function(e){var d=e.target.result;['records','pending','cache'].forEach(function(s){if(!d.objectStoreNames.contains(s))d.createObjectStore(s,{keyPath:s==='cache'?'key':'osbCode'});});};r.onsuccess=function(e){db=e.target.result;res(db);};r.onerror=rej;});}
function dbPut(s,d){return new Promise(function(res,rej){if(!db){res(false);return;}var t=db.transaction([s],'readwrite'),st=t.objectStore(s),r=st.put(d);r.onsuccess=function(){res(true);};r.onerror=rej;});}
function dbGet(s,k){return new Promise(function(res,rej){if(!db){res(null);return;}var t=db.transaction([s],'readonly'),st=t.objectStore(s),r=st.get(k);r.onsuccess=function(e){res(e.target.result);};r.onerror=rej;});}
function dbGetAll(s){return new Promise(function(res,rej){if(!db){res([]);return;}var t=db.transaction([s],'readonly'),st=t.objectStore(s),r=st.getAll();r.onsuccess=function(e){res(e.target.result||[]);};r.onerror=rej;});}
function dbDelete(s,k){return new Promise(function(res,rej){if(!db){res(false);return;}var t=db.transaction([s],'readwrite'),st=t.objectStore(s),r=st.delete(k);r.onsuccess=function(){res(true);};r.onerror=rej;});}

function autoFillLatLng(){
  var village=document.getElementById('village').value;
  var city=document.getElementById('city').value;
  var province=document.getElementById('province').value;
  var indicator=document.getElementById('locationIndicator');
  if(!province){document.getElementById('lat').value='';document.getElementById('lng').value='';indicator.style.display='none';return;}
  var loc=null, level='';
  if(village&&sliceData.locationByVillage&&sliceData.locationByVillage[village]){loc=sliceData.locationByVillage[village];level='ບ້ານ '+village;}
  else if(city&&sliceData.locationByCity&&sliceData.locationByCity[city]){loc=sliceData.locationByCity[city];level='ເມືອງ '+city;}
  else if(province&&sliceData.locationByProvince&&sliceData.locationByProvince[province]){loc=sliceData.locationByProvince[province];level='ແຂວງ '+province;}
  if(loc){
    document.getElementById('lat').value=loc.lat;document.getElementById('lng').value=loc.lng;
    indicator.className='location-indicator found';indicator.style.display='flex';
    indicator.innerHTML='<i class="fa fa-map-marker"></i> ພິກັດຈາກ: '+level+' ('+loc.lat.toFixed(4)+', '+loc.lng.toFixed(4)+')';return;
  }
  if(!isOnline){indicator.className='location-indicator not-found';indicator.style.display='flex';indicator.innerHTML='<i class="fa fa-exclamation-triangle"></i> ບໍ່ພົບພິກັດ (Offline)';return;}
  indicator.className='location-indicator loading';indicator.style.display='flex';indicator.innerHTML='<i class="fa fa-spinner fa-spin"></i> ກຳລັງດຶງພິກັດ...';
  google.script.run.withSuccessHandler(function(r){
    if(r.success){document.getElementById('lat').value=r.lat;document.getElementById('lng').value=r.lng;indicator.className='location-indicator found';indicator.innerHTML='<i class="fa fa-map-marker"></i> ພິກັດລະດັບ '+r.level+' ('+r.lat.toFixed(4)+', '+r.lng.toFixed(4)+')';}
    else{indicator.className='location-indicator not-found';indicator.innerHTML='<i class="fa fa-exclamation-triangle"></i> ບໍ່ພົບພິກັດໃນ Slice Sheet';}
  }).withFailureHandler(function(){indicator.className='location-indicator not-found';indicator.innerHTML='<i class="fa fa-times"></i> ດຶງພິກັດລົ້ມເຫຼວ';}).getLocationData(province,city,village);
}

function switchView(view){
  currentView=view;
  var views=['card','map','table','chart','display','stats'];
  views.forEach(function(v){
    document.getElementById(v+'View').style.display=(v===view)?'block':'none';
    document.getElementById('tab-'+v).classList.toggle('active',v===view);
  });
  if(view==='map'){initOrRefreshMap();}
  else if(view==='table'){renderTable();}
  else if(view==='chart'){renderCharts();}
  else if(view==='display'){renderDisplay();}
  else if(view==='stats'){renderStats();}
}

function applyFilters(){
  var filters={
    province:document.getElementById('filterProvince').value,
    city:document.getElementById('filterCity').value,
    village:document.getElementById('filterVillage').value,
    status:document.getElementById('filterStatus').value
  };
  showInfo('ກຳລັງໂຫຼດ...','loading');
  document.getElementById('spinIcon').classList.add('fa-spin');
  if(!isOnline){
    dbGetAll('records').then(function(all){
      document.getElementById('spinIcon').classList.remove('fa-spin');
      var filtered=all.filter(function(r){
        if(filters.province!=='ALL'&&r.province!==filters.province)return false;
        if(filters.city!=='ALL'&&r.city!==filters.city)return false;
        if(filters.village!=='ALL'&&r.village!==filters.village)return false;
        if(filters.status!=='ALL'&&r.status!==filters.status)return false;
        return true;
      });
      allLoadedData=filtered;refreshCurrentView(filtered,true);
    });
    return;
  }
  google.script.run.withSuccessHandler(function(r){
    document.getElementById('spinIcon').classList.remove('fa-spin');
    if(r.success){
      allLoadedData=r.data;r.data.forEach(function(rec){dbPut('records',rec);});refreshCurrentView(r.data,false);
    } else {
      showInfo(r.message,'error');
      document.getElementById('cardContainer').innerHTML='<div style="text-align:center;padding:50px;color:#999;">'+r.message+'</div>';
    }
  }).withFailureHandler(function(e){document.getElementById('spinIcon').classList.remove('fa-spin');showError(e);}).getFilteredData(filters);
}

  else if(currentView==='chart')renderCharts();
  else if(currentView==='display')renderDisplay();
  else if(currentView==='stats')renderStats();
}

function openPanel(name) {
  var panelEl = document.getElementById('panel'+name.charAt(0).toUpperCase()+name.slice(1));
  var sideEl  = document.getElementById('mapSidePanels');
  // hide all panels
  ['panelRadius','panelLasso','panelRouting'].forEach(function(id) {
    var p=document.getElementById(id); if(p) p.classList.remove('active');
  });
  if(activeSidePanel===name) {
    // toggle off
    sideEl.classList.remove('open');
    activeSidePanel=null;
  } else {
    activeSidePanel=name;
    if(panelEl) panelEl.classList.add('active');
    sideEl.classList.add('open');
    setTimeout(function(){if(leafletMap)leafletMap.invalidateSize();},320);
  }
}
function closePanel(name) {
  openPanel(name); // toggles it off
}

function initOrRefreshMap() {
  if (!leafletMap) {
    leafletMap=L.map('leafletMap',{zoomControl:true,preferCanvas:true}).setView([18.0,103.0],7);
    mapTileLayer=L.tileLayer(TILE_URLS.street.url,{attribution:TILE_URLS.street.attr}).addTo(leafletMap);
  }
  mapFilteredData=applyMapFilterLocal();
  refreshMapDisplay();
  setTimeout(function(){leafletMap.invalidateSize();},200);
}

function applyMapFilterLocal(){
  var el=document.getElementById('mapFilterStatus');
  var status=el?el.value:'ALL';
  if(status==='ALL')return allLoadedData.slice();
  return allLoadedData.filter(function(r){return r.status===status;});
}
function applyMapFilter(){mapFilteredData=applyMapFilterLocal();refreshMapDisplay();}

function refreshMapDisplay(){
  if(!leafletMap)return;
  clearAllMapLayers();
  var data=mapFilteredData.length?mapFilteredData:applyMapFilterLocal();

}

function clearAllMapLayers(){

  if(heatLayer){try{leafletMap.removeLayer(heatLayer);}catch(e){}heatLayer=null;}
  personMarkerLayers=[];
}

  var bounds=[];
  Object.values(groups).forEach(function(g){
    var n=g.members.length;

function renderHeatmap(data){
  var points=data.filter(function(r){return r.lat&&r.lng;}).map(function(r){return[parseFloat(r.lat),parseFloat(r.lng),0.8];});
  if(!points.length)return;
  if(typeof L.heatLayer==='function'){
    heatLayer=L.heatLayer(points,{radius:40,blur:30,gradient:{0.2:'#0000ff',0.4:'#00ff00',0.6:'#ffff00',1.0:'#ff0000'}}).addTo(leafletMap);
    var lats=points.map(function(p){return p[0];}),lngs=points.map(function(p){return p[1];});
    leafletMap.fitBounds([[Math.min.apply(null,lats),Math.min.apply(null,lngs)],[Math.max.apply(null,lats),Math.max.apply(null,lngs)]],{padding:[40,40]});
  } else {
    var groups={};
    data.forEach(function(r){if(!r.lat||!r.lng)return;var k=r.village||r.city||'x';if(!groups[k])groups[k]={lat:parseFloat(r.lat),lng:parseFloat(r.lng),count:0};groups[k].count++;});
    var maxC=Math.max.apply(null,Object.values(groups).map(function(g){return g.count;}));
    var bounds=[];
    Object.values(groups).forEach(function(g){var t=g.count/maxC;var col=t>0.7?'#ff2200':t>0.4?'#ffaa00':t>0.2?'#aaff00':'#0066ff';L.circleMarker([g.lat,g.lng],{radius:Math.max(30,Math.min(100,20+g.count*15)),color:'transparent',fillColor:col,fillOpacity:0.45}).addTo(leafletMap);bounds.push([g.lat,g.lng]);});
    if(bounds.length)leafletMap.fitBounds(bounds,{padding:[40,40]});
    showToast('info','Heatmap','ໃຊ້ Fallback Mode',3000);
  }
  setTimeout(function(){leafletMap.invalidateSize();},200);
}

function haversine(lat1,lon1,lat2,lon2){var R=6371,dLat=(lat2-lat1)*Math.PI/180,dLon=(lon2-lon1)*Math.PI/180;var a=Math.sin(dLat/2)*Math.sin(dLat/2)+Math.cos(lat1*Math.PI/180)*Math.cos(lat2*Math.PI/180)*Math.sin(dLon/2)*Math.sin(dLon/2);return R*2*Math.atan2(Math.sqrt(a),Math.sqrt(1-a));}

function calculateRoute(){
  if(routingStops.length<2){showToast('warning','ຕ້ອງການ ≥ 2 ຈຸດ','',3000);return;}
  var stops=routingStops.slice(),ordered=[stops.shift()];
  while(stops.length){var last=ordered[ordered.length-1],bestIdx=0,bestDist=Infinity;stops.forEach(function(s,i){var d=haversine(last.lat,last.lng,s.lat,s.lng);if(d<bestDist){bestDist=d;bestIdx=i;}});ordered.push(stops.splice(bestIdx,1)[0]);}
  routingLayers.forEach(function(l){try{leafletMap.removeLayer(l);}catch(e){}});routingLayers=[];
  var latlngs=ordered.map(function(s){return[s.lat,s.lng];});
  var line=L.polyline(latlngs,{color:'#232C8B',weight:3,dashArray:'8,6',opacity:0.85}).addTo(leafletMap);routingLayers.push(line);
  ordered.forEach(function(s,i){var icon=L.divIcon({className:'',html:'<div style="background:#232C8B;color:white;border-radius:50%;width:24px;height:24px;display:flex;align-items:center;justify-content:center;font-size:11px;font-weight:700;border:2px solid white;box-shadow:0 2px 6px rgba(0,0,0,0.3);">'+(i+1)+'</div>',iconSize:[24,24],iconAnchor:[12,12]});var m=L.marker([s.lat,s.lng],{icon:icon}).addTo(leafletMap);m.bindTooltip(s.name||'',{permanent:true,direction:'top'});routingLayers.push(m);});
  var totalKm=0;for(var i=0;i<ordered.length-1;i++)totalKm+=haversine(ordered[i].lat,ordered[i].lng,ordered[i+1].lat,ordered[i+1].lng);
  var distEl=document.getElementById('routingDistance');if(distEl){distEl.style.display='block';distEl.innerHTML='📐 ລວມ '+totalKm.toFixed(1)+' km | '+ordered.length+' ຈຸດ';}
  routingStops=ordered;updateRoutingPanel();leafletMap.fitBounds(line.getBounds(),{padding:[30,30]});
  showToast('success','Route','ລວມ '+totalKm.toFixed(1)+' km',4000);
}

  var btn=document.getElementById('btnGPS');if(btn){btn.classList.add('active');btn.innerHTML='⏳';}

    if(btn){btn.classList.remove('active');btn.innerHTML='📍 ຕຳແໜ່ງ';}
    if(gpsMarker)leafletMap.removeLayer(gpsMarker);if(gpsCircle)leafletMap.removeLayer(gpsCircle);
    gpsCircle=L.circle([lat,lng],{radius:acc,color:'#2196F3',fillColor:'#2196F3',fillOpacity:0.08,weight:1.5}).addTo(leafletMap);
    var icon=L.divIcon({className:'',html:'<div style="width:20px;height:20px;position:relative;"><div style="width:14px;height:14px;background:#2196F3;border-radius:50%;position:absolute;top:3px;left:3px;border:2px solid white;box-shadow:0 0 0 0 rgba(33,150,243,0.4);animation:gpsPulse 2s infinite;"></div></div>',iconSize:[20,20],iconAnchor:[10,10]});
    gpsMarker=L.marker([lat,lng],{icon:icon,zIndexOffset:9999}).addTo(leafletMap);
    gpsMarker.bindPopup('<b>📍 ຕຳແໜ່ງຂ້ອຍ</b><br>±'+Math.round(acc)+'m').openPopup();
    leafletMap.setView([lat,lng],13);
    var nearest=null,nearestDist=Infinity;
    mapFilteredData.forEach(function(r){if(!r.lat||!r.lng)return;var d=haversine(lat,lng,parseFloat(r.lat),parseFloat(r.lng));if(d<nearestDist){nearestDist=d;nearest=r;}});
    if(nearest)showToast('success','OSB ໃກ້ທີ່ສຸດ',(nearest.fullName||'')+' ('+nearestDist.toFixed(2)+' km)',6000);
    else showToast('info','📍 ຕຳແໜ່ງ','±'+Math.round(acc)+'m',4000);
  },function(err){if(btn){btn.classList.remove('active');btn.innerHTML='📍 ຕຳແໜ່ງ';}showToast('error','GPS',err.message,4000);},{enableHighAccuracy:true,timeout:10000,maximumAge:0});
}

  return allLoadedData.filter(function(r){
    if(prov!=='ALL'&&r.province!==prov)return false;if(city!=='ALL'&&r.city!==city)return false;if(village!=='ALL'&&r.village!==village)return false;

function onLoginSuccess(user,showWelcome){

  var avatar=document.getElementById('userAvatar');

  else{avatar.src='https://ui-avatars.com/api/?name='+encodeURIComponent(user.fullName||'U')+'&background=4A5FC1&color=fff';}

function doRegister(){
  var username=document.getElementById('regUsername').value.trim(),password=document.getElementById('regPassword').value,fullName=document.getElementById('regFullName').value.trim(),province=document.getElementById('regProvince').value,city=document.getElementById('regCity').value;
  if(!username||!password||!fullName||!province){showAuthMsg('registerMsg','error','⚠️ ກະລຸນາປ້ອນຂໍ້ມູນທີ່ຈຳເປັນ (*)');return;}
  if(password.length<6){showAuthMsg('registerMsg','error','⚠️ ລະຫັດຜ່ານ ≥ 6 ຕົວ');return;}
  var btn=document.getElementById('registerBtn');btn.disabled=true;btn.innerHTML='<i class="fa fa-spinner fa-spin"></i> ກຳລັງດຳເນີນການ...';
  var imgFile=document.getElementById('regIdCard').files[0];
  function sendReg(imgData){
    google.script.run.withSuccessHandler(function(r){btn.disabled=false;btn.innerHTML='<i class="fa fa-user-plus"></i> ສົ່ງຄຳຂໍ';if(r.success){showAuthMsg('registerMsg','success','✅ '+r.message);setTimeout(function(){switchAuthTab('login');},3000);}else{showAuthMsg('registerMsg','error','❌ '+r.message);}}).withFailureHandler(function(){btn.disabled=false;btn.innerHTML='<i class="fa fa-user-plus"></i> ສົ່ງຄຳຂໍ';showAuthMsg('registerMsg','error','❌ ຜິດພາດ');}).registerUser({username:username,password:password,fullName:fullName,province:province,city:city,profileImgData:imgData||null});
  }
  if(imgFile){var reader=new FileReader();reader.onload=function(e){sendReg(e.target.result);};reader.readAsDataURL(imgFile);}else{sendReg(null);}
}

var adminCurrentTab='pending';
function openAdminPanel(){if(!currentUser||!currentUser.isAdmin)return;document.getElementById('adminPanelModal').style.display='block';switchAdminTab('pending');}
function closeAdminPanel(){document.getElementById('adminPanelModal').style.display='none';}
function switchAdminTab(tab){
  adminCurrentTab=tab;
  document.querySelectorAll('.admin-panel-tab').forEach(function(t,i){t.classList.toggle('active',(i===0&&tab==='pending')||(i===1&&tab==='users'));});
  document.getElementById('adminTabPending').style.display=tab==='pending'?'block':'none';document.getElementById('adminTabUsers').style.display=tab==='users'?'block':'none';
  if(tab==='pending')loadPendingUsers();else loadAllUsers();
}
function loadPendingUsers(){
  document.getElementById('adminTabPending').innerHTML='<div style="text-align:center;padding:30px;color:#aaa;"><i class="fa fa-spinner fa-spin" style="font-size:28px;"></i></div>';
  google.script.run.withSuccessHandler(function(r){
    if(!r.success){document.getElementById('adminTabPending').innerHTML='<p style="color:red;">'+r.message+'</p>';return;}
    document.getElementById('pendingBadge').textContent=r.data.length?' ('+r.data.length+')':'';
    if(!r.data.length){document.getElementById('adminTabPending').innerHTML='<div style="text-align:center;padding:40px;color:#aaa;"><i class="fa fa-check-circle" style="font-size:48px;color:#4CAF50;"></i><p>ບໍ່ມີຄຳຂໍ</p></div>';return;}
    var html='';r.data.forEach(function(u){var img=u.profileImg?getImageUrl(u.profileImg):'https://ui-avatars.com/api/?name='+encodeURIComponent(u.fullName||'U')+'&background=232C8B&color=fff&size=100';html+='<div class="user-card"><img src="'+img+'" onerror="this.src=\'https://ui-avatars.com/api/?name=U&background=232C8B&color=fff&size=100\'"><div class="user-card-info"><div class="name">'+(u.fullName||u.username)+'</div><div class="meta">👤 '+u.username+' | 📍 '+(u.province||'-')+'</div></div><div class="user-card-actions"><button class="w3-button w3-green w3-small" onclick="approveUser(\''+u.username+'\')"><i class="fa fa-check"></i></button><button class="w3-button w3-red w3-small" onclick="rejectUser(\''+u.username+'\')"><i class="fa fa-times"></i></button></div></div>';});
    document.getElementById('adminTabPending').innerHTML=html;
  }).getPendingUsers();
}
function loadAllUsers(){
  document.getElementById('adminTabUsers').innerHTML='<div style="text-align:center;padding:30px;color:#aaa;"><i class="fa fa-spinner fa-spin" style="font-size:28px;"></i></div>';
  google.script.run.withSuccessHandler(function(r){
    if(!r.success)return;
    var html='';r.data.forEach(function(u){var img=u.profileImg?getImageUrl(u.profileImg):'https://ui-avatars.com/api/?name='+encodeURIComponent(u.fullName||'U')+'&background=232C8B&color=fff&size=100';var sc=u.status==='ອະນຸມັດ'?'#4CAF50':'#FF9800';html+='<div class="user-card"><img src="'+img+'" onerror="this.src=\'https://ui-avatars.com/api/?name=U&background=232C8B&color=fff&size=100\'"><div class="user-card-info"><div class="name">'+(u.fullName||u.username)+'</div><div class="meta">👤 '+u.username+' | <span style="color:'+sc+';font-weight:600;">'+u.status+'</span></div></div><div class="user-card-actions">'+(u.role!=='ຜູ້ດູແລລະບົບ'?'<button class="w3-button w3-indigo w3-small" onclick="promoteAdmin(\''+u.username+'\')"><i class="fa fa-star"></i></button>':'')+(u.status==='ອະນຸມັດ'?'<button class="w3-button w3-orange w3-small" onclick="suspendUser(\''+u.username+'\')"><i class="fa fa-ban"></i></button>':'<button class="w3-button w3-green w3-small" onclick="approveUser(\''+u.username+'\')"><i class="fa fa-unlock"></i></button>')+'</div></div>';});
    document.getElementById('adminTabUsers').innerHTML=html||'<p style="color:#aaa;text-align:center;">ບໍ່ມີ</p>';
  }).getAllUsers();
}
function approveUser(u){google.script.run.withSuccessHandler(function(r){showToast(r.success?'success':'error',r.success?'ອະນຸມັດ':'ຜິດ',r.message,4000);if(r.success)loadPendingUsers();}).updateUserStatus(u,'ອະນຸມັດ');}
function rejectUser(u){if(!confirm('ປະຕິເສດ '+u+'?'))return;google.script.run.withSuccessHandler(function(r){showToast(r.success?'warning':'error','ປະຕິເສດ',r.message,4000);if(r.success)loadPendingUsers();}).updateUserStatus(u,'ປະຕິເສດ');}
function suspendUser(u){if(!confirm('ລ໊ອກ '+u+'?'))return;google.script.run.withSuccessHandler(function(r){showToast('warning','ລ໊ອກ',r.message,4000);if(r.success)loadAllUsers();}).updateUserStatus(u,'ລໍຖ້າການອະນຸມັດ');}
function promoteAdmin(u){if(!confirm('ຍົກ '+u+' ເປັນ Admin?'))return;google.script.run.withSuccessHandler(function(r){showToast(r.success?'success':'error','ຍົກ',r.message,4000);if(r.success)loadAllUsers();}).updateUserRole(u,'ຜູ້ດູແລລະບົບ');}
