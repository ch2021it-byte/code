   function saveToSyncQueue(action, data) {
  if (!db) return;
  
  const syncItem = {
    action: action,
    data: data,
    timestamp: new Date().getTime(),
    retryCount: 0,
    notified: false
  };
  
  const transaction = db.transaction(['syncQueue'], 'readwrite');
  const store = transaction.objectStore('syncQueue');
  store.add(syncItem);
}

   function handleSyncFailure(item) {
  const tx = db.transaction(['syncQueue'], 'readwrite');
  const store = tx.objectStore('syncQueue');
  item.retryCount = (item.retryCount || 0) + 1;
  
  if (item.retryCount >= 5 && !item.notified) {
    item.notified = true;
    const actionNames = {
      save: 'ບັນທຶກຄອບຄົວ',
      deleteMember: 'ລົບສະມາຊິກ',
      deleteHousehold: 'ລົບຄອບຄົວ'
    };
    showToast('⚠️ ບໍ່ສາມາດ' + (actionNames[item.action] || 'Sync') + 'ໄດ້ ກະລຸນາກວດສອບອິນເຕີເນັດ ຫຼື Refresh ໜ້າເວັບ', 'error');
  }
  
  store.put(item);
}

function processSyncQueue() {
  if (!db) return;
  
  const transaction = db.transaction(['syncQueue'], 'readonly');
  const store = transaction.objectStore('syncQueue');
  const request = store.getAll();
  
  request.onsuccess = function() {
    const items = request.result;
    
    items.forEach(item => {
      if (item.action === 'save') {
        google.script.run
          .withSuccessHandler(function(result) {
            if (result.success) {
              const tx = db.transaction(['syncQueue'], 'readwrite');
              tx.objectStore('syncQueue').delete(item.id);
            } else {
              handleSyncFailure(item);
            }
          })
          .withFailureHandler(function() {
            handleSyncFailure(item);
          })
          .saveHousehold(item.data, villageSheetId);
      } else if (item.action === 'deleteMember') {
        google.script.run
          .withSuccessHandler(function(result) {
            if (result.success) {
              const tx = db.transaction(['syncQueue'], 'readwrite');
              tx.objectStore('syncQueue').delete(item.id);
            } else {
              handleSyncFailure(item);
            }
          })
          .withFailureHandler(function() {
            handleSyncFailure(item);
          })
          .deleteMember(item.data.memberCode, item.data.reason, villageSheetId);
      } else if (item.action === 'deleteHousehold') {
        google.script.run
          .withSuccessHandler(function(result) {
            if (result.success) {
              const tx = db.transaction(['syncQueue'], 'readwrite');
              tx.objectStore('syncQueue').delete(item.id);
            } else {
              handleSyncFailure(item);
            }
          })
          .withFailureHandler(function() {
            handleSyncFailure(item);
          })
          .deleteHousehold(item.data.householdCode, item.data.reason, villageSheetId);
      }
    });
  };
}

 function showLoading(targetCount = 0, message = 'ກຳລັງໂຫລດຂໍ້ມູນ...') {
  const loadingDiv = document.getElementById('loading');
  const counterDiv = document.getElementById('loadingCounter');
  const messageDiv = document.getElementById('loadingMessage');
  
  loadingDiv.classList.add('active');
  
  if (counterDiv) {
    counterDiv.textContent = '0';
  }
  
  if (messageDiv) {
    messageDiv.textContent = message;
  }
  
  // Animation counter
  if (targetCount > 0 && counterDiv) {
    let current = 0;
    const duration = 2000; // 2 วินาที
    const increment = targetCount / (duration / 50);
    
    const timer = setInterval(() => {
      current += increment;
      if (current >= targetCount) {
        current = targetCount;
        clearInterval(timer);
      }
      counterDiv.textContent = Math.floor(current);
    }, 50);
    
    // เก็บ timer ไว้เพื่อ clear ตอน hide
    loadingDiv.setAttribute('data-timer', timer);
  }
}

 function hideLoading() {
  const loadingDiv = document.getElementById('loading');
  
  // Clear timer ถ้ามี
  const timer = loadingDiv.getAttribute('data-timer');
  if (timer) {
    clearInterval(parseInt(timer));
    loadingDiv.removeAttribute('data-timer');
  }
  
  loadingDiv.classList.remove('active');
}

let map = null;
let marker = null;
let selectedLatLng = null;
let streetLayer = null;
let satelliteLayer = null;
let currentLayer = 'satellite';

function openLocationPicker() {
  document.getElementById('locationModal').classList.add('active');
  
  // Destroy old map if exists
  if (map) {
    map.remove();
    map = null;
  }
  
  // Wait for modal to be visible
  setTimeout(() => {
    initializeMap();
  }, 300);
}
function initializeMap() {
  // Default to Vientiane, Laos
  const defaultLat = 17.9757;
  const defaultLng = 102.6331;
  
  // Get current value if exists
  const currentValue = document.getElementById('housePosition').value;
  let initialLat = defaultLat;
  let initialLng = defaultLng;
  
  if (currentValue && currentValue.includes(',')) {
    const parts = currentValue.split(',');
    initialLat = parseFloat(parts[0].trim()) || defaultLat;
    initialLng = parseFloat(parts[1].trim()) || defaultLng;
  }
  
  // Create map
  map = L.map('map', {
    center: [initialLat, initialLng],
    zoom: 15,
    preferCanvas: true
  });
  
  // Create Street Map Layer
  streetLayer = L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
    attribution: '© OpenStreetMap',
    maxZoom: 19
  });
  
  // Create Satellite Layer
  satelliteLayer = L.tileLayer('https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}', {
    attribution: '© Esri',
    maxZoom: 19
  });
  
  // Add satellite layer as default
  satelliteLayer.addTo(map);
  
  // Create custom beautiful marker
  const beautifulIcon = createCustomMarkerIcon();
  
  // Add marker with custom icon
  marker = L.marker([initialLat, initialLng], { 
    draggable: true,
    icon: beautifulIcon
  }).addTo(map);
  
  selectedLatLng = { lat: initialLat, lng: initialLng };
  updateLocationDisplay();
  
  // Update location when marker is dragged
  marker.on('dragend', function(e) {
    const pos = marker.getLatLng();
    selectedLatLng = { lat: pos.lat, lng: pos.lng };
    updateLocationDisplay();
  });
  
  // Update location when map is clicked
  map.on('click', function(e) {
    marker.setLatLng(e.latlng);
    selectedLatLng = { lat: e.latlng.lat, lng: e.latlng.lng };
    updateLocationDisplay();
  });
  
  // Force tiles to load
  map.invalidateSize();
  
  // Try to get user location
  if (navigator.geolocation) {
    navigator.geolocation.getCurrentPosition(
      function(position) {
        const lat = position.coords.latitude;
        const lng = position.coords.longitude;
        map.setView([lat, lng], 17);
        marker.setLatLng([lat, lng]);
        selectedLatLng = { lat: lat, lng: lng };
        updateLocationDisplay();
        showToast('📍 ພົບຕຳແໜ່ງປັດຈຸບັນຂອງທ່ານ', 'success');
      },
      function(error) {
        console.log('Geolocation error:', error);
      }
    );
  }
}

function switchMapLayer(layerType) {
  if (!map) return;
  
  // Remove all layers
  if (streetLayer && map.hasLayer(streetLayer)) {
    map.removeLayer(streetLayer);
  }
  if (satelliteLayer && map.hasLayer(satelliteLayer)) {
    map.removeLayer(satelliteLayer);
  }
  
  // Add selected layer
  if (layerType === 'street') {
    streetLayer.addTo(map);
    currentLayer = 'street';
  } else {
    satelliteLayer.addTo(map);
    currentLayer = 'satellite';
  }
  
  // Update button states
  document.getElementById('btnStreetMap').classList.toggle('active', layerType === 'street');
  document.getElementById('btnSatellite').classList.toggle('active', layerType === 'satellite');
  
  // Force redraw
  setTimeout(() => {
    map.invalidateSize();
  }, 100);
}

// ✅ ใหม่: ตัดเบอร์โทรดิบจากฐานข้อมูล ให้เหลือ 8 หลักสุดท้าย สำหรับใส่ในช่องกรอก
function extractPhoneDigits(fullPhone) {
  if (!fullPhone) return '';
  const digitsOnly = String(fullPhone).replace(/\D/g, '');
  if (!digitsOnly) return '';
  return digitsOnly.slice(-8);
}

// ✅ ใหม่: เติมรหัส 85620 นำหน้าเลข 8 หลักที่กรอก ก่อนบันทึกลงฐานข้อมูล
function buildFullPhone(digits) {
  const clean = (digits || '').replace(/\D/g, '').slice(0, 8);
  return clean ? ('85620' + clean) : '';
}

// ✅ ใหม่: จัดเรียงฟอร์มสมาชิกใน DOM ตามสายพัวพัน (หัวหน้าครอบครัว > เมีย/ผัว > ลูก > ...)
function resortMemberFormsInDOM() {
  const container = document.getElementById('membersContainer');
  if (!container) return;
  const sections = Array.from(container.querySelectorAll('.member-section'));
  if (sections.length === 0) return;

  sections.sort(function(a, b) {
    const relA = a.querySelector('.memberRelationship') ? a.querySelector('.memberRelationship').value : '';
    const relB = b.querySelector('.memberRelationship') ? b.querySelector('.memberRelationship').value : '';
    const rankA = getRelationshipRank(relA);
    const rankB = getRelationshipRank(relB);
    if (rankA !== rankB) return rankA - rankB;
    const nameA = a.querySelector('.memberName') ? a.querySelector('.memberName').value : '';
    const nameB = b.querySelector('.memberName') ? b.querySelector('.memberName').value : '';
    return nameA.localeCompare(nameB, 'lo');
  });

  sections.forEach(function(section) {
    container.appendChild(section);
  });
}

let scaleX = 1;
let scaleY = 1;

function openCropModal(file, previewId) {
  if (!file || !file.type.startsWith('image/')) {
    showToast('ກະລຸນາເລືອກໄຟລ์ຮູບພາບ', 'error');
    return;
  }
  
  currentCropPreviewId = previewId;
  const reader = new FileReader();
  
  reader.onload = function(e) {
    const cropImage = document.getElementById('cropImage');
    cropImage.src = e.target.result;
    
    // Destroy previous cropper if exists
    if (cropper) {
      cropper.destroy();
    }
    
    // Initialize Cropper.js
    cropper = new Cropper(cropImage, {
      aspectRatio: 1, // Default 1:1
      viewMode: 2,
      dragMode: 'move',
      autoCropArea: 0.8,
      restore: false,
      guides: true,
      center: true,
      highlight: false,
      cropBoxMovable: true,
      cropBoxResizable: true,
      toggleDragModeOnDblclick: false,
      background: false,
      responsive: true,
      checkOrientation: true
    });
    
    // Reset scales
    scaleX = 1;
    scaleY = 1;
    
    // Show modal
    document.getElementById('cropModal').classList.add('active');
  };
  
  reader.readAsDataURL(file);
}

function loadExistingRegistryData() {
  // Load from form if exists
  const bookCount = parseInt(document.getElementById('registryBookCount').value) || 0;
  const registryNumber = document.getElementById('registryNumber').value;
  
  if (bookCount > 0 && registryNumber) {
    // Parse existing data
    const parts = registryNumber.split('|');
    
    // Parse old registry
    if (parts.length >= 1 && parts[0].includes(':')) {
      const oldPart = parts[0].split(':')[1].trim();
      if (oldPart) {
        const oldItems = oldPart.split(',').map(n => n.trim()).filter(n => n);
        const oldNumbers = [];
        const oldDates = [];
        
        oldItems.forEach(item => {
          const itemParts = item.split(' (');
          oldNumbers.push(itemParts[0]);
          if (itemParts[1]) {
            oldDates.push(itemParts[1].replace(')', ''));
          } else {
            oldDates.push('');
          }
        });
        
        registryData.old.enabled = true;
        registryData.old.count = oldNumbers.length;
        registryData.old.numbers = oldNumbers;
        registryData.old.dates = oldDates;
        
        document.getElementById('chkOldRegistry').checked = true;
        document.getElementById('oldRegistryCount').value = oldNumbers.length;
        updateOldRegistryNumbers();
        toggleRegistryOption('old');
      }
    }
    
    // Parse new registry
    if (parts.length >= 2 && parts[1].includes(':')) {
      const newPart = parts[1].split(':')[1].trim();
      if (newPart) {
        const newItems = newPart.split(',').map(n => n.trim()).filter(n => n);
        const newNumbers = [];
        const newDates = [];
        
        newItems.forEach(item => {
          const itemParts = item.split(' (');
          newNumbers.push(itemParts[0]);
          if (itemParts[1]) {
            newDates.push(itemParts[1].replace(')', ''));
          } else {
            newDates.push('');
          }
        });
        
        registryData.new.enabled = true;
        registryData.new.count = newNumbers.length;
        registryData.new.numbers = newNumbers;
        registryData.new.dates = newDates;
        
        document.getElementById('chkNewRegistry').checked = true;
        document.getElementById('newRegistryCount').value = newNumbers.length;
        updateNewRegistryNumbers();
        toggleRegistryOption('new');
      }
    }
  }
  
  updateRegistrySummary();
}

function toggleRegistryOption(type) {
  const checkbox = document.getElementById(`chk${type === 'old' ? 'Old' : type === 'new' ? 'New' : 'No'}Registry`);
  const details = document.getElementById(`${type}RegistryDetails`);
  const option = document.getElementById(`${type}RegistryOption`);
  
  if (type === 'none') {
    registryData.none = checkbox.checked;
    if (checkbox.checked) {
      option.classList.add('selected');
      // Uncheck other options
      document.getElementById('chkOldRegistry').checked = false;
      document.getElementById('chkNewRegistry').checked = false;
      registryData.old.enabled = false;
      registryData.new.enabled = false;
      document.getElementById('oldRegistryOption').classList.remove('selected');
      document.getElementById('newRegistryOption').classList.remove('selected');
      document.getElementById('oldRegistryDetails').classList.remove('active');
      document.getElementById('newRegistryDetails').classList.remove('active');
    } else {
      option.classList.remove('selected');
    }
  } else {
    const enabled = checkbox.checked;
    registryData[type].enabled = enabled;
    
    if (enabled) {
      option.classList.add('selected');
      details.classList.add('active');
      // Uncheck "none"
      document.getElementById('chkNoRegistry').checked = false;
      registryData.none = false;
      document.getElementById('noRegistryOption').classList.remove('selected');
    } else {
      option.classList.remove('selected');
      details.classList.remove('active');
      registryData[type].count = 0;
      registryData[type].numbers = [];
    }
  }
  
  updateRegistrySummary();
}

function confirmRegistry() {
  // Collect all registry numbers with dates
  const oldItems = [];
  const newItems = [];
  
  if (registryData.old.enabled) {
    for (let i = 0; i < registryData.old.count; i++) {
      const number = registryData.old.numbers[i] || '';
      const date = registryData.old.dates[i] || '';
      if (number.trim()) {
        if (date && date !== '//') {
          oldItems.push(`${number.trim()} (${date})`);
        } else {
          oldItems.push(number.trim());
        }
      }
    }
  }
  
  if (registryData.new.enabled) {
    for (let i = 0; i < registryData.new.count; i++) {
      const number = registryData.new.numbers[i] || '';
      const date = registryData.new.dates[i] || '';
      if (number.trim()) {
        if (date && date !== '//') {
          newItems.push(`${number.trim()} (${date})`);
        } else {
          newItems.push(number.trim());
        }
      }
    }
  }
  
  // Calculate total
  const totalBooks = oldItems.length + newItems.length;
  
  // Format registry number string
  let registryNumberText = '';
  if (oldItems.length > 0) {
    registryNumberText += 'ເກົ່າ: ' + oldItems.join(', ');
  }
  if (newItems.length > 0) {
    if (registryNumberText) registryNumberText += ' | ';
    registryNumberText += 'ໃໝ່: ' + newItems.join(', ');
  }
  
  // Update form fields
  document.getElementById('registryBookCount').value = totalBooks;
  document.getElementById('registryNumber').value = registryNumberText;
  
  // Show success message
  showToast(`ບັນທຶກສຳເລັດ: ${totalBooks} ປື້ມ`, 'success');
  
  closeRegistryModal();
}

// ฟังก์ชันจัดการ Multi-Select
function initializeMultiSelect() {
  const diseaseTextareas = document.querySelectorAll('.multi-select-field');
  
  diseaseTextareas.forEach(textarea => {
    // สร้าง container สำหรับแสดง tags
    const tagsContainer = document.createElement('div');
    tagsContainer.className = 'disease-tags-container';
    tagsContainer.style.cssText = 'display: flex; flex-wrap: wrap; gap: 5px; margin-bottom: 10px;';
    
    textarea.parentElement.insertBefore(tagsContainer, textarea);
    
    // ถ้ามีข้อมูลเดิม แสดงเป็น tags
    if (textarea.value) {
      const diseases = textarea.value.split(',').map(d => d.trim()).filter(d => d);
      diseases.forEach(disease => {
        addDiseaseTag(disease, tagsContainer, textarea);
      });
    }
    
    // Event listener สำหรับ Enter
    textarea.addEventListener('keydown', function(e) {
      if (e.key === 'Enter') {
        e.preventDefault();
        const value = this.value.trim();
        
        if (value) {
          addDiseaseTag(value, tagsContainer, this);
          this.value = '';
        }
      }
    });
  });
}

function addDiseaseTag(disease, container, textarea) {
  const tag = document.createElement('span');
  tag.className = 'disease-tag';
  tag.style.cssText = 'background: #e3f2fd; color: #1976d2; padding: 5px 10px; border-radius: 15px; display: inline-flex; align-items: center; gap: 5px; font-size: 0.9rem;';
  tag.innerHTML = `
    ${disease}
    <span onclick="removeTag(this)" style="cursor: pointer; font-weight: bold; margin-left: 5px;">×</span>
  `;
  
  container.appendChild(tag);
  
  // อัปเดต textarea value
  updateDiseaseTextarea(textarea);
}

function removeTag(element) {
  const tag = element.parentElement;
  const container = tag.parentElement;
  const textarea = container.nextElementSibling;
  
  tag.remove();
  updateDiseaseTextarea(textarea);
}

function updateDiseaseTextarea(textarea) {
  const container = textarea.previousElementSibling;
  const tags = container.querySelectorAll('.disease-tag');
  const diseases = Array.from(tags).map(tag => tag.textContent.trim().replace('×', '').trim());
  
  textarea.value = diseases.join(', ');
}

// แทนที่ฟังก์ชัน handleDiseaseInput เดิมด้วยโค้ดนี้
function handleDiseaseInput(event, memberId) {
  const input = document.getElementById(`diseaseInput${memberId}`);
  const value = input.value.trim();
  
  // กด Enter เพื่อเพิ่มโรค
  if (event.key === 'Enter') {
    event.preventDefault();
    
    if (value) {
      addDiseaseTag(value, memberId);
      input.value = '';
    }
  }
  
  // Tab เพื่อเพิ่มโรค
  if (event.key === 'Tab' && value) {
    event.preventDefault();
    addDiseaseTag(value, memberId);
    input.value = '';
  }
}
// ฟังก์ชันเพื่อจับการเลือกจาก datalist (เพิ่มใหม่)
function setupDiseaseInputListeners(memberId) {
  const input = document.getElementById(`diseaseInput${memberId}`);
  if (!input) return;
  
  // เมื่อเลือกจาก datalist
  input.addEventListener('change', function() {
    const value = this.value.trim();
    if (value) {
      addDiseaseTag(value, memberId);
      this.value = '';
    }
  });
  
  // เมื่อ blur (ออกจาก input) และมีค่า
  input.addEventListener('blur', function() {
    setTimeout(() => {
      const value = this.value.trim();
      if (value) {
        addDiseaseTag(value, memberId);
        this.value = '';
      }
    }, 200);
  });
}
// แทนที่ฟังก์ชัน addDiseaseTag เดิมด้วยโค้ดนี้
function addDiseaseTag(disease, memberId) {
  const container = document.getElementById(`diseaseTagsContainer${memberId}`);
  const hiddenTextarea = document.getElementById(`diseaseHidden${memberId}`);
  
  if (!container || !hiddenTextarea) return;
  
  // ลบ placeholder ถ้ามี
  const placeholder = container.querySelector('.empty-placeholder');
  if (placeholder) {
    placeholder.remove();
  }
  
  // ตรวจสอบว่ามีโรคนี้แล้วหรือยัง
  const existingTags = container.querySelectorAll('.disease-tag');
  for (let tag of existingTags) {
    if (tag.getAttribute('data-disease') === disease) {
      showToast('⚠️ ມີພະຍາດນີ້ແລ້ວ', 'error');
      return;
    }
  }
  
  // สร้าง tag
  const tag = document.createElement('span');
  tag.className = 'disease-tag';
  tag.setAttribute('data-disease', disease);
  tag.style.cssText = 'background: linear-gradient(135deg, #e3f2fd 0%, #bbdefb 100%); color: #1976d2; padding: 8px 12px; border-radius: 20px; display: inline-flex; align-items: center; gap: 8px; font-size: 0.9rem; box-shadow: 0 2px 5px rgba(0,0,0,0.1); transition: all 0.3s; animation: slideIn 0.3s ease;';
  tag.innerHTML = `
    <span style="font-weight: 500;">🔴 ${disease}</span>
    <span onclick="removeDiseaseTag(this, ${memberId})" 
          style="cursor: pointer; font-weight: bold; font-size: 1.2rem; color: #f44336; transition: transform 0.2s; width: 20px; height: 20px; display: flex; align-items: center; justify-content: center; border-radius: 50%; background: rgba(244,67,54,0.1);"
          onmouseover="this.style.transform='scale(1.3)'; this.style.background='rgba(244,67,54,0.2)'"
          onmouseout="this.style.transform='scale(1)'; this.style.background='rgba(244,67,54,0.1)'">×</span>
  `;
  
  container.appendChild(tag);
  
  // อัปเดต hidden textarea
  updateDiseaseTextarea(memberId);
  
  showToast('✅ ເພີ່ມພະຍາດສຳເລັດ', 'success');
}

function updateMobilePaginationUI(totalPages) {
  document.getElementById('currentMobilePage').textContent = currentMobilePage;
  document.getElementById('totalMobilePages').textContent = totalPages;
  
  document.getElementById('btnMobileFirst').disabled = currentMobilePage === 1;
  document.getElementById('btnMobilePrev').disabled = currentMobilePage === 1;
  document.getElementById('btnMobileNext').disabled = currentMobilePage === totalPages;
  document.getElementById('btnMobileLast').disabled = currentMobilePage === totalPages;
  
  document.getElementById('mobilePagination').style.display = totalPages > 1 ? 'flex' : 'none';
}

function changeMobilePage(direction) {
  const totalPages = Math.ceil(totalMobileRows / mobileRowsPerPage);
  
  switch(direction) {
    case 'first':
      currentMobilePage = 1;
      break;
    case 'prev':
      if (currentMobilePage > 1) currentMobilePage--;
      break;
    case 'next':
      if (currentMobilePage < totalPages) currentMobilePage++;
      break;
    case 'last':
      currentMobilePage = totalPages;
      break;
  }
  
  loadMobileData();
  document.getElementById('mobileTable').scrollIntoView({ behavior: 'smooth', block: 'start' });

// Search Mobile Data
// ===================================
function searchMobileData() {
  const searchTerm = document.getElementById('searchMobileInput').value.trim();
  
  if (!searchTerm) {
    loadMobileData();
    return;
  }
  
  if (searchTerm.length < 2) {
    showToast('ກະລຸນາພິມຢ່າງໜ້ອຍ 2 ຕົວອັກສອນ', 'error');
    return;
  }
  
  showLoading();
  
  google.script.run
    .withSuccessHandler(function(results) {
      hideLoading();
      
      if (!results || results.length === 0) {
        const tbody = document.getElementById('mobileTableBody');
        tbody.innerHTML = `
          <tr>
            <td colspan="7" style="text-align: center; padding: 40px;">
              <div style="font-size: 48px; margin-bottom: 10px;">🔍</div>
              <div>ບໍ່ພົບຂໍ້ມູນທີ່ຄົ້ນຫາ: "${searchTerm}"</div>
              <button onclick="clearMobileSearch()" class="btn btn-primary" style="margin-top: 15px;">
                ❌ ລ້າງການຄົ້ນຫາ
              </button>
            </td>
          </tr>
        `;
        document.getElementById('mobilePagination').style.display = 'none';
        showToast('ບໍ່ພົບຜົນການຄົ້ນຫາ', 'error');
        return;
      }
      
      displayMobileSearchResults(results, searchTerm);
      showToast(`ພົບ ${results.length} ລາຍການ`, 'success');
    })
    .withFailureHandler(function(error) {
      hideLoading();
      showToast('ເກີດຂໍ້ຜິດພາດໃນການຄົ້ນຫາ: ' + error.message, 'error');
    })
    .searchMobile(searchTerm, villageSheetId);
}

function filterHouseholdList() {
  const searchInput = document.getElementById('householdSearchInput');
  if (!searchInput) return;
  
  const searchValue = searchInput.value.trim().toLowerCase();
  const cards = document.querySelectorAll('.household-card');
  
  cards.forEach(card => {
    const nameEl = card.querySelector('h4');
    const phoneEl = card.querySelector('p');
    const name = nameEl ? nameEl.textContent.toLowerCase() : '';
    const phone = phoneEl ? phoneEl.textContent.toLowerCase() : '';
    
    const isMatch = !searchValue || name.includes(searchValue) || phone.includes(searchValue);
    card.style.display = isMatch ? 'block' : 'none';
  });
}

  const memberDiv = document.getElementById(`mobile-member-${id}`);

function reorderMobileMembers() {
  const container = document.getElementById('mobileMembersContainer');
  const memberDivs = container.querySelectorAll('.mobile-member-section');
  
  memberDivs.forEach(function(div, index) {
    const newNumber = index + 1;
    const heading = div.querySelector('h3');
    if (heading) {
      heading.setAttribute('data-member-number', newNumber);
      const span = heading.querySelector('span');
      if (span) {
        span.textContent = '👤 ສະມາຊິກຄົນທີ່ ' + newNumber;
      }
    }
    div.id = 'mobile-member-' + newNumber;
  });
  
  mobileMemberCounter = memberDivs.length;
}

function calculateEndDate(memberId, periodId) {
  const startDateInput = document.getElementById('rentalStart' + memberId + '-' + periodId);
  const daysInput = document.getElementById('rentalDays' + memberId + '-' + periodId);
  const endDateDisplay = document.getElementById('rentalEnd' + memberId + '-' + periodId);
  
  if (!startDateInput || !daysInput || !endDateDisplay) return;
  
  const startDate = startDateInput.value;
  const days = parseInt(daysInput.value);
  
  if (startDate && days > 0) {
    const start = new Date(startDate);
    const end = new Date(start);
    end.setDate(start.getDate() + days);
    
    const endDateStr = end.toISOString().split('T')[0];
    endDateDisplay.textContent = 'ສິ້ນສຸດ: ' + endDateStr;
    endDateDisplay.setAttribute('data-end-date', endDateStr);
  } else {
    endDateDisplay.textContent = '-';
    endDateDisplay.removeAttribute('data-end-date');
  }
}

function getRelationshipRank(rel) {
  const cleanRel = (rel || '').toString().trim().normalize('NFC');
  if (!cleanRel) return 999;

  // 1) ຫົວໜ້າຄອບຄົວ ต้องตรงเป๊ะทั้งข้อความเท่านั้น (ไม่ใช่แค่มีคำนี้ปนอยู่ในคำอื่น)
  if (cleanRel === 'ຫົວໜ້າຄອບຄົວ') return 0;

  // 2) ຜົວ/ເມຍ ຫົວໜ້າຄອບຄົວ (เช่น "ເມຍຫົວໜ້າຄອບຄົວ", "ຜົວຫົວໜ້າຄອບຄົວ")
  const spouseKeywords = ['ເມຍ', 'ຜົວ', 'ພັນລະຍາ', 'ສາມີ'];
  if (spouseKeywords.some(function(k) { return cleanRel.indexOf(k) !== -1; })) return 1;

  // 3) ລູກເຂີຍ / ລູກໃພ້ (ต้องเช็คก่อน "ลูก" เฉยๆ เพราะมีคำว่า "ลูก" ปนอยู่ข้างใน)
  const childInLawKeywords = ['ລູກເຂີຍ', 'ລູກໃພ້'];
  if (childInLawKeywords.some(function(k) { return cleanRel.indexOf(k) !== -1; })) return 3;

  // 4) ລູກ
  if (cleanRel.indexOf('ລູກ') !== -1) return 2;

  // 5) ຫຼານ / ຫລານ
  if (cleanRel.indexOf('ຫຼານ') !== -1 || cleanRel.indexOf('ຫລານ') !== -1) return 4;

  // 6) ພໍ່ / ແມ່
  if (cleanRel.indexOf('ພໍ່') !== -1 || cleanRel.indexOf('ແມ່') !== -1) return 5;

  // 7) ພີ່ນ້ອງ
  if (cleanRel.indexOf('ພີ່ນ້ອງ') !== -1) return 6;

  // นอกเหนือจากนี้ ให้อยู่ท้ายสุด
  return 999;
}

function sortMembersByRelationship(members) {
  return [...members].sort((a, b) => {
    const rankA = getRelationshipRank(a.relationship);
    const rankB = getRelationshipRank(b.relationship);
    if (rankA !== rankB) return rankA - rankB;
    return (a.name || '').localeCompare(b.name || '', 'lo');
  });
}

function calculateAgeFromBirthDate(birthDateStr) {
  if (!birthDateStr) return null;
  const parts = birthDateStr.split('/');
  if (parts.length !== 3) return null;
  const day = parseInt(parts[0]), month = parseInt(parts[1]), year = parseInt(parts[2]);
  if (!day || !month || !year || isNaN(day) || isNaN(month) || isNaN(year)) return null;
  const birth = new Date(year, month - 1, day);
  const today = new Date();
  let age = today.getFullYear() - birth.getFullYear();
  const m = today.getMonth() - birth.getMonth();
  if (m < 0 || (m === 0 && today.getDate() < birth.getDate())) age--;
  return age >= 0 ? age : null;
}

function isAgeInRange(age, range) {
  if (age === null) return false;
  if (range.min !== '' && age < parseInt(range.min)) return false;
  if (range.max !== '' && age > parseInt(range.max)) return false;
  return true;
}
function fieldPassesFilter(item, colKey, filterState, rangeFields, multiFields, textFields) {
  const filterVal = filterState[colKey];
  if (filterVal === undefined || filterVal === null) return true;

if (colKey === 'age') {
    // ✅ อายุใช้แค่ "สร้างคอลัมน์ + ติ๊กถูก" ในตาราง ไม่ใช้กรองสมาชิกออกจากรายชื่อ
    return true;
  }

  if (rangeFields.indexOf(colKey) !== -1) {
    if (!filterVal || (filterVal.min === '' && filterVal.max === '')) return true;
    const num = parseFloat(item[colKey]);
    if (filterVal.min !== '' && (isNaN(num) || num < parseFloat(filterVal.min))) return false;
    if (filterVal.max !== '' && (isNaN(num) || num > parseFloat(filterVal.max))) return false;
    return true;
  }

if (multiFields.indexOf(colKey) !== -1) {
    if (!filterVal || filterVal.length === 0) return true;
    if (colKey === 'disease') {
      const itemDisease = item.disease || '';
      return filterVal.some(function(token) { return itemDisease.indexOf(token) !== -1; });
    }
    return filterVal.indexOf(item[colKey]) !== -1;
  }

  if (textFields && textFields.indexOf(colKey) !== -1) {
    if (filterVal === '') return true;
    const itemText = String(item[colKey] || '').toLowerCase();
    return itemText.indexOf(String(filterVal).toLowerCase()) !== -1;
  }

  if (filterVal === '') return true;
  return String(item[colKey] || '') === filterVal;
}

function getUniqueValuesForColumn(dataArray, key) {
  const set = new Set();
  dataArray.forEach(function(item) {
    const val = item[key];
    if (val !== undefined && val !== null && String(val).trim() !== '') {
      set.add(String(val).trim());
    }
  });
  const values = Array.from(set);
  if (key === 'unit') {
    return values.sort(naturalUnitCompare);
  }
  return values.sort(function(a,b){ return a.localeCompare(b, 'lo'); });
}

function getUniqueDiseaseTokens(members) {
  const set = new Set();
  members.forEach(function(m) {
    if (!m.disease) return;
    m.disease.split(',').forEach(function(d) {
      const t = d.trim();
      if (t) set.add(t);
    });
  });
  return Array.from(set).sort(function(a,b){ return a.localeCompare(b,'lo'); });
}

function applyNewFilters() {
  const body = document.getElementById('filterFieldsBody');

  body.querySelectorAll('[data-multiselect="1"]').forEach(function(wrap) {
    const scope = wrap.getAttribute('data-scope');
    const key = wrap.getAttribute('data-key');
    const checkedBoxes = wrap.querySelectorAll('input[type="checkbox"]:checked');
    const values = Array.from(checkedBoxes).map(function(cb) { return cb.value; });
    if (values.length === 0) {
      delete summaryFilters[scope][key];
    } else {
      summaryFilters[scope][key] = values;
    }
  });

body.querySelectorAll('select:not([data-multiselect="1"])').forEach(function(select) {
    const scope = select.getAttribute('data-scope');
    const key = select.getAttribute('data-key');
    if (select.value === '') {
      delete summaryFilters[scope][key];
    } else {
      summaryFilters[scope][key] = select.value;
    }
  });

  body.querySelectorAll('input[data-textfilter="1"]').forEach(function(input) {
    const scope = input.getAttribute('data-scope');
    const key = input.getAttribute('data-key');
    const val = input.value.trim();
    if (val === '') {
      delete summaryFilters[scope][key];
    } else {
      summaryFilters[scope][key] = val;
    }
  });

body.querySelectorAll('input[data-range]').forEach(function(input) {
    const scope = input.getAttribute('data-scope');
    const key = input.getAttribute('data-key');
    const rangeType = input.getAttribute('data-range');
    if (!summaryFilters[scope][key] || typeof summaryFilters[scope][key] !== 'object' || Array.isArray(summaryFilters[scope][key])) {
      summaryFilters[scope][key] = { min: '', max: '' };
    }
    summaryFilters[scope][key][rangeType] = input.value;
    if (summaryFilters[scope][key].min === '' && summaryFilters[scope][key].max === '') {
      delete summaryFilters[scope][key];
    }
  });

  body.querySelectorAll('.age-ranges-list').forEach(function(listEl) {
    const scope = listEl.getAttribute('data-scope');
    const rows = listEl.querySelectorAll('.age-range-row');
    const ranges = [];
    rows.forEach(function(rowEl) {
      const minInput = rowEl.querySelector('input[data-age-range="min"]');
      const maxInput = rowEl.querySelector('input[data-age-range="max"]');
      const min = minInput ? minInput.value : '';
      const max = maxInput ? maxInput.value : '';
      if (min !== '' || max !== '') {
        ranges.push({ min: min, max: max });
      }
    });
    if (ranges.length === 0) {
      delete summaryFilters[scope].age;
    } else {
      summaryFilters[scope].age = ranges;
    }
  });

  updateActiveFilterCount();
  closeFilterFieldsModal();
  renderSummaryContent();
  showToast('นำใช้ตัวกรองสำเร็จ', 'success');
}

function updateActiveFilterCount() {
  let count = 0;
  ['household','member'].forEach(function(scope) {
    Object.keys(summaryFilters[scope]).forEach(function(key) {
      const val = summaryFilters[scope][key];
      if (Array.isArray(val)) {
        if (val.length > 0) count++;
      } else if (val && typeof val === 'object') {
        if (val.min !== '' || val.max !== '') count++;
      } else if (val && val !== '') {
        count++;
      }
    });
  });
  const badge = document.getElementById('activeFilterCount');
  if (badge) badge.textContent = count;
}
function getFilteredHouseholdSummary() {
  const result = [];

  // ✅ "หน่วย" ถือเป็นคุณสมบัติระดับครอบครัวเสมอ ไม่ว่าจะไปติ๊กเลือกจากฝั่งหัวหน้าครอบครัว
  // หรือฝั่งสมาชิกก็ตาม เพื่อกันปัญหาสมาชิกหายเวลากรองด้วยหน่วย
  const effectiveUnitFilter =
    (summaryFilters.household.unit && summaryFilters.household.unit.length > 0) ? summaryFilters.household.unit :
    (summaryFilters.member.unit && summaryFilters.member.unit.length > 0) ? summaryFilters.member.unit :
    null;

  const householdFilterKeys = Object.keys(summaryFilters.household).filter(function(k){ return k !== 'unit'; });
  const memberFilterKeys = Object.keys(summaryFilters.member).filter(function(k){ return k !== 'unit'; });

  const householdFilterActive = householdFilterKeys.length > 0;
  const memberFilterActive = memberFilterKeys.length > 0;

  const applyHouseholdFilter = (currentFilterScope === 'household' || currentFilterScope === 'both') && householdFilterActive;
  const applyMemberFilter = (currentFilterScope === 'member' || currentFilterScope === 'both') && memberFilterActive;

  householdsData.forEach(function(household) {
    // ✅ กรอง "หน่วย" ที่ระดับครอบครัวก่อนเสมอ
    if (effectiveUnitFilter && effectiveUnitFilter.indexOf(household.unit) === -1) return;

    if (applyHouseholdFilter && !householdPassesFilters(household)) return;

    let matchingMembers = household.members || [];
    if (applyMemberFilter) {
      matchingMembers = matchingMembers.filter(memberPassesFiltersExcludingUnit);
    }

    result.push({
      household: household,
      members: sortMembersByRelationship(matchingMembers)
    });
  });
  return result;
}

function renderAgeChartControls() {
  const list = document.getElementById('ageChartControlsList');
  if (!list) return;
  const ranges = customAgeChartRanges || getDefaultAgeChartRanges();
  list.innerHTML = ranges.map(function(r) { return buildAgeChartRangeRowHTML(r); }).join('');
}

function addAgeChartRangeRow() {
  const list = document.getElementById('ageChartControlsList');
  if (!list) return;
  list.insertAdjacentHTML('beforeend', buildAgeChartRangeRowHTML({ label: '', min: '', max: '' }));
}

function removeAgeChartRangeRow(btn) {
  const row = btn.closest('.age-chart-range-row');
  if (row) row.remove();
}

function applyAgeChartRanges() {
  const list = document.getElementById('ageChartControlsList');
  if (!list) return;
  const rows = list.querySelectorAll('.age-chart-range-row');
  const newRanges = [];
  rows.forEach(function(row) {
    const label = row.querySelector('.acr-label').value.trim();
    const min = row.querySelector('.acr-min').value;
    const max = row.querySelector('.acr-max').value;
    if (min !== '' && max !== '') {
      newRanges.push({
        label: label || (min + '-' + max),
        min: parseInt(min),
        max: parseInt(max)
      });
    }
  });
  if (newRanges.length === 0) {
    showToast('ກະลุนาระบุอย่างน้อย 1 ช่วงอายุ', 'error');
    return;
  }
  customAgeChartRanges = newRanges;
  redrawAgeChartOnly();
  showToast('นำใช้ช่วงอายุใหม่สำเร็จ', 'success');
}

function resetAgeChartRanges() {
  customAgeChartRanges = null;
  renderAgeChartControls();
  redrawAgeChartOnly();
  showToast('คืนค่าช่วงอายุเริ่มต้นสำเร็จ', 'success');
}

function redrawAgeChartOnly() {
  let members = [];
  if (summaryDataType === 'household') {
    getFilteredHouseholdSummary().forEach(item => { members = members.concat(item.members); });
  } else {
    members = getFilteredMobileSummary();
  }
  renderAgeChart(members);
}

function renderAgeChart(members) {
  if (ageChartInstance) {
    ageChartInstance.destroy();
    ageChartInstance = null;
  }
  const canvas = document.getElementById('chartAge');
  if (!canvas) return;

  const ranges = customAgeChartRanges || getDefaultAgeChartRanges();
  const counts = {};
  ranges.forEach(function(r) { counts[r.label] = 0; });
  counts['ไม่ระบุ'] = 0;

  members.forEach(function(m) {
    const age = calculateAgeFromBirthDate(m.birthDate);
    if (age === null) { counts['ไม่ระบุ']++; return; }
    let matched = false;
    for (let i = 0; i < ranges.length; i++) {
      if (age >= ranges[i].min && age <= ranges[i].max) {
        counts[ranges[i].label]++;
        matched = true;
        break;
      }
    }
    if (!matched) counts['ไม่ระบุ']++;
  });

const entries = Object.entries(counts);
  ageChartInstance = new Chart(canvas, {
    type: 'bar',
    data: { labels: entries.map(e => e[0]), datasets: [{ label: 'จำนวน (คน)', data: entries.map(e => e[1]), backgroundColor: '#667eea' }] },
    options: { responsive: true, plugins: { legend: { display: false } } }
  });
  chartInstancesMap['chartAge'] = ageChartInstance;
}

  XLSX.utils.book_append_sheet(wb, XLSX.utils.json_to_sheet(Object.entries(countByGlobal(members,'education')).map(function(e){ return {'การศึกษา': e[0], 'จำนวน': e[1]}; })), 'การศึกษา');

  XLSX.utils.book_append_sheet(wb, XLSX.utils.json_to_sheet(Object.entries(countByGlobal(members,'ethnicity')).map(function(e){ return {'ชนเผ่า': e[0], 'จำนวน': e[1]}; })), 'ชนเผ่า');
  XLSX.utils.book_append_sheet(wb, XLSX.utils.json_to_sheet(Object.entries(countByGlobal(members,'occupation')).map(function(e){ return {'อาชีพ': e[0], 'จำนวน': e[1]}; })), 'อาชีพ');

  const ranges = customAgeChartRanges || getDefaultAgeChartRanges();
  const ageCounts = {};
  ranges.forEach(function(r){ ageCounts[r.label] = 0; });
  ageCounts['ไม่ระบุ'] = 0;
  members.forEach(function(m) {
    const age = calculateAgeFromBirthDate(m.birthDate);
    if (age === null) { ageCounts['ไม่ระบุ']++; return; }
    let matched = false;
    for (let i=0;i<ranges.length;i++) {
      if (age >= ranges[i].min && age <= ranges[i].max) { ageCounts[ranges[i].label]++; matched = true; break; }
    }
    if (!matched) ageCounts['ไม่ระบุ']++;
  });
  XLSX.utils.book_append_sheet(wb, XLSX.utils.json_to_sheet(Object.entries(ageCounts).map(function(e){ return {'ช่วงอายุ': e[0], 'จำนวน': e[1]}; })), 'ช่วงอายุ');
}

async function downloadPDF() {

  showDownloadProgress('ກຳລັງສ້າງ PDF...', '📄');

  let container = null;
  let renderDiv = null;

  try {
    if (typeof html2canvas === 'undefined' || !window.jspdf || !window.jspdf.jsPDF) {
      throw new Error('ບໍ່ພົบ library ທີ່ຈຳເປັນ ກະລຸນາ Refresh ໜ້າເວັບແລ້ວລອງໃໝ່');
    }

    updateDownloadProgress(5, 'ກຳລັງກະກຽມຂໍ້ມູນ...');
    await new Promise(r => setTimeout(r, 50));

    const reportHTML = generateReportHTMLForActiveTab();

    container = document.createElement('div');
    container.innerHTML = reportHTML;
    container.style.cssText = 'position: absolute; left: -9999px; top: 0; width: 1400px; background: white; margin: 0; padding: 0;';
    document.body.appendChild(container);

    updateDownloadProgress(15, 'ກຳລັງກະກຽມ Layout...');

    if (document.fonts && document.fonts.ready) {
      await document.fonts.ready;
    }
    await new Promise(r => requestAnimationFrame(() => requestAnimationFrame(r)));
    await new Promise(r => setTimeout(r, 300));

    const rootDiv = container.firstElementChild || container;

    const headerEls = [];
    const bodyEls = [];
    Array.from(rootDiv.children).forEach(function(el) {
      if ((el.tagName === 'H1' || el.tagName === 'P') && bodyEls.length === 0) {
        headerEls.push(el);
      } else {
        bodyEls.push(el);
      }
    });

    let chunks = [];
    if (bodyEls.length === 1 && bodyEls[0].tagName === 'TABLE') {
      chunks = splitTableIntoChunks(bodyEls[0], 40);
    } else if (bodyEls.length === 1 && bodyEls[0].querySelector && bodyEls[0].querySelector('table')) {
      const innerTable = bodyEls[0].querySelector('table');
      chunks = splitTableIntoChunks(innerTable, 40);
    } else {
      const BLOCKS_PER_CHUNK = 6;
      for (let i = 0; i < bodyEls.length; i += BLOCKS_PER_CHUNK) {
        chunks.push(bodyEls.slice(i, i + BLOCKS_PER_CHUNK));
      }
    }
    if (chunks.length === 0) chunks = [bodyEls];

    updateDownloadProgress(20, 'ກຳລັງແບ່ງໜ້າ PDF (' + chunks.length + ' ຊຸດ)...');

    renderDiv = document.createElement('div');
    renderDiv.style.cssText = 'position: absolute; left: -9999px; top: 0; width: 1400px; background: white; padding: 15px; font-family: "Phetsarath OT","Noto Sans Lao",Arial,sans-serif;';
    document.body.appendChild(renderDiv);

const margin = 10;
    const jsPDFCtor = window.jspdf.jsPDF;
    const pdf = new jsPDFCtor({ unit: 'mm', format: 'a4', orientation: 'landscape' });
    const usableWidth = pdf.internal.pageSize.getWidth() - margin * 2;

    let pagesAdded = 0;
    let skippedChunks = 0;

    for (let i = 0; i < chunks.length; i++) {
      renderDiv.innerHTML = '';

      if (i === 0) {
        headerEls.forEach(function(el) { renderDiv.appendChild(el.cloneNode(true)); });
      }

      const chunkEls = chunks[i];
      if (Array.isArray(chunkEls)) {
        chunkEls.forEach(function(el) { renderDiv.appendChild(el.cloneNode(true)); });
      } else if (chunkEls) {
        renderDiv.appendChild(chunkEls.cloneNode(true));
      }

      await new Promise(r => setTimeout(r, 30));

      try {
        const canvas = await html2canvas(renderDiv, {
          scale: 1.5,
          useCORS: true,
          backgroundColor: '#ffffff',
          logging: false
        });

        if (!canvas || !canvas.width || !canvas.height) {
          console.warn('ຂ້າມຊຸດທີ່ ' + (i + 1) + ' — canvas ວ່າງເປົ່າ');
          skippedChunks++;
        } else {
const imgData = canvas.toDataURL('image/jpeg', 0.95);
          let imgHeightMM = (canvas.height * usableWidth) / canvas.width;
          if (!imgHeightMM || isNaN(imgHeightMM) || imgHeightMM <= 0) {
            imgHeightMM = usableWidth * 0.6;
          }

          if (pagesAdded > 0) pdf.addPage();
          pdf.addImage(imgData, 'JPEG', margin, margin, usableWidth, imgHeightMM);
          pagesAdded++;
        }
      } catch (chunkError) {
        console.error('ຂ້າມຊຸດທີ່ ' + (i + 1) + ' ເນື່ອງຈາກ error:', chunkError);
        skippedChunks++;
      }

      const pct = 20 + Math.round(((i + 1) / chunks.length) * 70);
      updateDownloadProgress(pct, 'ກຳລັງສ້າງໜ້າ PDF... (' + (i + 1) + '/' + chunks.length + ')');
    }

    if (pagesAdded === 0) {
      throw new Error('ບໍ່ສາมารถสร้างหน้า PDF ได้เลยแม้แต่หน้าเดียว');
    }

    updateDownloadProgress(95, 'ກຳລັງບັນທຶກໄฟล์...');
    await new Promise(r => setTimeout(r, 100));

    const fileName = 'สรุปข้อมูล_' + new Date().getTime() + '.pdf';
    pdf.save(fileName);

    updateDownloadProgress(100, 'ดาวโหลดสำเร็จ!');
    if (skippedChunks > 0) {
      showToast('ดาวน์โหลด PDF สำเร็จ (ข้าม ' + skippedChunks + ' ชุดที่มีปัญหา)', 'success');
    } else {
      showToast('ดาวน์โหลด PDF สำเร็จ', 'success');
    }

  } catch (error) {
    console.error('PDF export error:', error);
    showToast('เกิดข้อผิดพลาดในการสร้าง PDF: ' + error.message, 'error');
  } finally {
    if (container && container.parentNode) container.parentNode.removeChild(container);
    if (renderDiv && renderDiv.parentNode) renderDiv.parentNode.removeChild(renderDiv);
    hideDownloadProgress();
    if (btn) { btn.disabled = false; btn.style.opacity = '1'; btn.style.cursor = 'pointer'; }
  }
}

function enableMarkerEdit(code, type) {
  const entry = gpsMarkersData[code];
  if (!entry) {
    showToast('ບໍ່ພົບ Marker', 'error');
    return;
  }

  // ปิดโหมดลากของ marker ก่อนหน้า (ถ้ามี) เพื่อให้ลากได้ทีละจุด
  if (currentEditingMarker && currentEditingMarker.marker) {
    currentEditingMarker.marker.dragging.disable();
  }

  const marker = entry.marker;
  marker.dragging.enable();
  currentEditingMarker = entry;

  marker.closePopup();
  showToast('🖐️ ສາມາດລາກ Marker ໄດ້ແລ້ວ! ລາກໄປບ່ອນທີ່ຖືກຕ້ອງ ແລ້ວປ່ອຍມືເພື່ອບັນທຶກອັດຕະໂນມັດ', 'success');

  marker.off('dragend');
  marker.on('dragend', function() {
    const pos = marker.getLatLng();
    const newPositionStr = pos.lat.toFixed(6) + ', ' + pos.lng.toFixed(6);
    saveMarkerNewPosition(code, type, newPositionStr, marker);
  });
}

function saveMarkerNewPosition(code, type, newPositionStr, marker) {
  showLoading(0, 'ກຳລັງບັນທຶກຕໍາແໜ່ງໃໝ່...');

  const successHandler = function(result) {
    hideLoading();
    if (result.success) {
      showToast('✅ ບັນທຶກຕໍາແໜ່ງໃໝ່ສຳເລັດ', 'success');

      if (type === 'household') {
        const h = householdsData.find(function(x) { return x.householdCode === code; });
        if (h) {
          h.housePosition = newPositionStr;
          saveToIndexedDB('households', h);
        }
      } else {
        const m = mobilesData.find(function(x) { return x.mobileCode === code; });
        if (m) {
          m.position = newPositionStr;
          saveToIndexedDB('mobiles', m);
        }
      }

      marker.dragging.disable();
      currentEditingMarker = null;
    } else {
      showToast('❌ ' + result.message, 'error');
    }
  };

  const failureHandler = function(error) {
    hideLoading();
    showToast('❌ ເກີດຂໍ້ຜິດພາດ: ' + error.message, 'error');
  };

if (type === 'household') {
    google.script.run
      .withSuccessHandler(successHandler)
      .withFailureHandler(failureHandler)
      .updateHouseholdPosition(code, newPositionStr, villageSheetId);
  } else {
    google.script.run
      .withSuccessHandler(successHandler)
      .withFailureHandler(failureHandler)
      .updateMobilePosition(code, newPositionStr, villageSheetId);
  }
}

// GPS REAL-TIME SEARCH (ค้นหาชื่อ/เบอร์โทร แบบเรียลไทม์)

function filterGpsMarkers(searchTerm) {
  gpsCurrentSearchTerm = (searchTerm || '').trim().toLowerCase();
  applyGpsFilters();
}

function applyGpsFilters() {
  if (!gpsSummaryMap) return;
  const term = gpsCurrentSearchTerm;
  const matchedBounds = [];

  Object.keys(gpsMarkersData).forEach(function(code) {
    const entry = gpsMarkersData[code];
    entry.marker.unbindTooltip();

    const unitMatch = gpsSelectedUnits.length === 0 || gpsSelectedUnits.indexOf(entry.unit) !== -1;

    let matchedItem = null;
    if (term) {
      (entry.matchItems || []).forEach(function(item) {
        if (matchedItem) return;
        const nameMatch = item.label && item.label.toLowerCase().indexOf(term) !== -1;
        const phoneMatch = item.phone && item.phone.indexOf(term) !== -1;
        if (nameMatch || phoneMatch) matchedItem = item;
      });
    }
    const searchMatch = !term || !!matchedItem;
    const visible = unitMatch && searchMatch;

    if (visible) {
      if (!gpsSummaryMap.hasLayer(entry.marker)) {
        entry.marker.addTo(gpsSummaryMap);
      }
      if (term && matchedItem) {
        entry.marker.bindTooltip(matchedItem.label, {
          permanent: true,
          direction: 'top',
          className: 'gps-search-label'
        }).openTooltip();
      }
      matchedBounds.push(entry.marker.getLatLng());
    } else {
      if (gpsSummaryMap.hasLayer(entry.marker)) {
        gpsSummaryMap.removeLayer(entry.marker);
      }
    }
  });

  if ((term || gpsSelectedUnits.length > 0) && matchedBounds.length > 0) {
    if (matchedBounds.length === 1) {
      gpsSummaryMap.setView(matchedBounds[0], 17);
    } else {
      gpsSummaryMap.fitBounds(matchedBounds, { padding: [60, 60] });
    }
  }
}

function clearGpsSearch() {
  const gpsSearchEl = document.getElementById('gpsSearchInput');
  if (gpsSearchEl) gpsSearchEl.value = '';
  gpsCurrentSearchTerm = '';
  applyGpsFilters();
}

// ===================================
// GPS UNIT FILTER (ตัวกรองหน่วย เลือกได้หลายรายการ)
// ===================================
function populateGpsUnitFilter() {
  const dropdown = document.getElementById('gpsUnitDropdown');
  if (!dropdown) return;

  gpsSelectedUnits = [];
  updateGpsUnitCountBadge();
  dropdown.innerHTML = '';

  const unitSet = new Set();
  if (summaryDataType === 'household') {
    householdsData.forEach(function(h) { if (h.unit) unitSet.add(h.unit); });
  } else {
    mobilesData.forEach(function(m) { if (m.unit) unitSet.add(m.unit); });
  }

  const sortedUnits = Array.from(unitSet).sort();

  if (sortedUnits.length === 0) {
    dropdown.innerHTML = '<div style="color:#999; padding:5px; text-align:center;">ບໍ່ມີຂໍ້ມູນໜ່ວຍ</div>';
    return;
  }

  sortedUnits.forEach(function(unit) {
    const label = document.createElement('label');
    label.style.cssText = 'display:flex; align-items:center; gap:8px; padding:8px 6px; cursor:pointer; border-bottom:1px solid #f0f0f0;';
    label.innerHTML = '<input type="checkbox" value="' + unit + '" onchange="toggleGpsUnit(\'' + unit + '\', this.checked)" style="width:18px; height:18px; cursor:pointer;"> <span>' + unit + '</span>';
    dropdown.appendChild(label);
  });
}

function toggleGpsUnitDropdown() {
  const dropdown = document.getElementById('gpsUnitDropdown');
  if (!dropdown) return;
  dropdown.style.display = (dropdown.style.display === 'none' || !dropdown.style.display) ? 'block' : 'none';
}

function toggleGpsUnit(unit, checked) {
  if (checked) {
    if (gpsSelectedUnits.indexOf(unit) === -1) gpsSelectedUnits.push(unit);
  } else {
    gpsSelectedUnits = gpsSelectedUnits.filter(function(u) { return u !== unit; });
  }
  updateGpsUnitCountBadge();
  applyGpsFilters();
}

function updateGpsUnitCountBadge() {
  const badge = document.getElementById('gpsUnitCount');
  if (badge) badge.textContent = gpsSelectedUnits.length;
}

function startMyLocationTracking() {
  if (!navigator.geolocation) {
    showToast('ບາວເຊີບໍ່ຮອງຮັບ GPS', 'error');
    return;
  }
  if (myLocationWatchId !== null) return;

  myLocationWatchId = navigator.geolocation.watchPosition(
    function(position) {
      const lat = position.coords.latitude;
      const lng = position.coords.longitude;

  if (!myLocationMarker) {
        const myIcon = L.divIcon({
          html: '<div style="font-size: 36px; line-height: 36px; text-align:center; filter: drop-shadow(0 3px 4px rgba(0,0,0,0.5));">📍</div>',
          className: '',
          iconSize: [36, 36],
          iconAnchor: [18, 36]
        });
        myLocationMarker = L.marker([lat, lng], { icon: myIcon, zIndexOffset: 1000 }).addTo(gpsSummaryMap);
        myLocationMarker.bindTooltip('📍 ຕໍາແໜ່ງຂອງຂ້ອຍ', { permanent: false });
      } else {
        myLocationMarker.setLatLng([lat, lng]);
      }
    },
    function(error) {
      console.error('Geolocation error:', error);
      showToast('ບໍ່ສາມາດເຂົ້າເຖິງຕໍາແໜ່ງ GPS ໄດ້: ' + error.message, 'error');
    },
    { enableHighAccuracy: true, maximumAge: 5000, timeout: 15000 }
  );

  const btn = document.getElementById('btnGpsMyLocationToggle');
  if (btn) btn.textContent = '🛑 ຢຸດຕິດຕາມຕໍາແໜ່ງ';
}

function stopMyLocationTracking() {
  if (myLocationWatchId !== null) {
    navigator.geolocation.clearWatch(myLocationWatchId);
    myLocationWatchId = null;
  }
  if (myLocationMarker && gpsSummaryMap) {
    gpsSummaryMap.removeLayer(myLocationMarker);
  }
  myLocationMarker = null;

  const btn = document.getElementById('btnGpsMyLocationToggle');
  if (btn) btn.textContent = '📍 ຕໍາແໜ່ງຂອງຂ້ອຍ';
}

function toggleMyLocationTracking() {
  if (myLocationWatchId !== null) {
    stopMyLocationTracking();
  } else {
    startMyLocationTracking();
  }
}
