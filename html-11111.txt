    function initIndexedDB() {

      request.onerror = function() {
        console.error('IndexedDB error:', request.error);
        showToast('ເກີດຂໍ້ຜິດພາດໃນການເປີດຖານຂໍ້ມູນ', 'error');
      };
      
      request.onsuccess = function() {
        db = request.result;
        loadInitialData();
      };
      
      request.onupgradeneeded = function(e) {
        db = e.target.result;
        
        if (!db.objectStoreNames.contains('loginData')) {
          db.createObjectStore('loginData', { keyPath: 'id' });
        }
        if (!db.objectStoreNames.contains('optionsData')) {
          db.createObjectStore('optionsData', { keyPath: 'id' });
        }
        if (!db.objectStoreNames.contains('households')) {
          db.createObjectStore('households', { keyPath: 'householdCode' });
        }
        if (!db.objectStoreNames.contains('mobiles')) {
  db.createObjectStore('mobiles', { keyPath: 'mobileCode' });
}
        if (!db.objectStoreNames.contains('syncQueue')) {
          const syncStore = db.createObjectStore('syncQueue', { keyPath: 'id', autoIncrement: true });
          syncStore.createIndex('timestamp', 'timestamp', { unique: false });
        }
      };
    }

function fetchLoginData() {

    console.warn('⚠️ Login data loading is taking longer than expected...');
    const messageDiv = document.getElementById('loadingMessage');

  google.script.run
    .withSuccessHandler(function(data) {
      clearTimeout(timeoutId);
      
      const endTime = new Date().getTime();
      const timeTaken = ((endTime - startTime) / 1000).toFixed(2);

function retryFetchLoginData(retryCount) {

    console.error('❌ Max retries reached');

    hideLoading();
    return;
  }
  
  console.log('🔄 Retry attempt: ' + (retryCount + 1) + '/' + maxRetries);
  
  const messageDiv = document.getElementById('loadingMessage');
  if (messageDiv) {
    messageDiv.textContent = '🔄 ລອງໃໝ່... ຄັ້ງທີ່ ' + (retryCount + 1);
  }
  
  setTimeout(function() {
    google.script.run
      .withSuccessHandler(function(data) {
        hideLoading();
        if (data && data.provinces) {
          loginData = data;
          populateLoginForm();
          showToast('ໂຫລດສຳເລັດ!', 'success');
        } else {
          retryFetchLoginData(retryCount + 1);
        }
      })
      .withFailureHandler(function(error) {
        console.error('Retry failed:', error);
        retryFetchLoginData(retryCount + 1);
      })
      .getLoginData();
  }, 2000 * (retryCount + 1)); // เพิ่มเวลารอทุกครั้ง
}

          console.error('>>> ERROR: Mobile data is not array');
          mobilesData = [];

          mobilesData = data;
          
          // บันทึกลง IndexedDB
          if (data.length > 0) {

        finishLoadingAllData();
      })
      .withFailureHandler(function(error) {
        console.error('>>> ERROR fetching mobile:', error);
        mobilesData = [];
        
        // ลองโหลดจาก IndexedDB
        loadFromIndexedDB('mobiles', null).then(data => {

          }

function finishLoadingAllData() {

    console.log('>>> Mobiles:', mobilesData.length);

      if (household.members && Array.isArray(household.members)) {
        totalPeople += household.members.length;
      }
    });
    totalPeople += mobilesData.length; // ประชากรเคลื่อนที่
    
    console.log('>>> Total people (all):', totalPeople);
    
    // แสดง Counter Animation
    const counterDiv = document.getElementById('loadingCounter');

    if (counterDiv && messageDiv && totalPeople > 0) {
      messageDiv.textContent = `ກຳລັງໂຫລດ ${totalPeople} ຄົນ...`;
      
      let current = 0;
      const duration = 1500; // 1.5 วินาที
      const increment = totalPeople / (duration / 30);
      
      const timer = setInterval(() => {
        current += increment;
        if (current >= totalPeople) {
          current = totalPeople;
          clearInterval(timer);
          
          // เปลี่ยนข้อความเป็น "ສຳເລັດ"
          messageDiv.textContent = `✅ ໂຫລດສຳເລັດ ${totalPeople} ຄົນ`;

    function saveToIndexedDB(storeName, data) {
      return new Promise((resolve, reject) => {
        const transaction = db.transaction([storeName], 'readwrite');
        const store = transaction.objectStore(storeName);
        const request = store.put(data);
        
        request.onsuccess = () => resolve(request.result);
        request.onerror = () => reject(request.error);
      });
    }

    function loadFromIndexedDB(storeName, key) {
      return new Promise((resolve, reject) => {
        const transaction = db.transaction([storeName], 'readonly');
        const store = transaction.objectStore(storeName);
        
        if (key) {
          const request = store.get(key);
          request.onsuccess = () => resolve(request.result ? request.result.data : null);
          request.onerror = () => reject(request.error);
        } else {
          const request = store.getAll();
          request.onsuccess = () => resolve(request.result);
          request.onerror = () => reject(request.error);
        }
      });
    }

    function deleteFromIndexedDB(storeName, key) {
      return new Promise((resolve, reject) => {
        const transaction = db.transaction([storeName], 'readwrite');
        const store = transaction.objectStore(storeName);
        const request = store.delete(key);
        
        request.onsuccess = () => resolve();
        request.onerror = () => reject(request.error);
      });
    }

function renderSearchDropdown(fieldId, items) {
  const dropdown = document.getElementById(fieldId + 'Dropdown');
  if (!dropdown) return;
  dropdown.innerHTML = '';
  if (!items || items.length === 0) {
    dropdown.innerHTML = '<div class="searchable-dropdown-empty">ບໍ່ພົບຂໍ້ມູນ</div>';
    return;
  }
  items.forEach(function(item) {
    const div = document.createElement('div');
    div.className = 'searchable-dropdown-item';
    div.textContent = item;
    div.onclick = function() { selectSearchDropdownItem(fieldId, item); };
    dropdown.appendChild(div);
  });
}

function openSearchDropdown(fieldId) {
  const input = document.getElementById(fieldId);
  if (input.disabled) return;
  
  document.querySelectorAll('.searchable-dropdown-list').forEach(function(el) {
    if (el.id !== fieldId + 'Dropdown') el.classList.remove('active');
  });
  
  renderSearchDropdown(fieldId, searchDropdownData[fieldId] || []);
  document.getElementById(fieldId + 'Dropdown').classList.add('active');
}

function filterSearchDropdown(fieldId) {
  const input = document.getElementById(fieldId);
  const searchTerm = input.value.trim().toLowerCase();
  const allItems = searchDropdownData[fieldId] || [];

  renderSearchDropdown(fieldId, filtered);

  document.getElementById(fieldId + 'Value').value = '';

function naturalUnitCompare(a, b) {
  const numA = parseFloat(String(a).replace(/[^\d.]/g, ''));
  const numB = parseFloat(String(b).replace(/[^\d.]/g, ''));
  const hasNumA = !isNaN(numA);
  const hasNumB = !isNaN(numB);
  if (hasNumA && hasNumB && numA !== numB) return numA - numB;
  return String(a).localeCompare(String(b), 'lo');
}

function getSortedUniqueUnits() {
  const unitSet = new Set();
  householdsData.forEach(h => { if (h.unit) unitSet.add(h.unit); });
  return Array.from(unitSet).sort(naturalUnitCompare);
}

function getUnitColor(unit, sortedUnits) {
  const idx = sortedUnits.indexOf(unit);
  const safeIdx = idx === -1 ? 0 : idx % UNIT_COLOR_PALETTE.length;
  return UNIT_COLOR_PALETTE[safeIdx];
}

function getSortedFilteredHouseholds() {
  let list = householdsData.slice();

  if (populationUnitFilter.length > 0) {
    list = list.filter(h => populationUnitFilter.indexOf(h.unit) !== -1);
  }

  switch (populationSortMode) {
    case 'unitDesc':
      list.sort((a, b) => naturalUnitCompare(b.unit || '', a.unit || ''));
      break;
    case 'nameAsc':
      list.sort((a, b) => (a.name || '').localeCompare(b.name || '', 'lo'));
      break;
    case 'nameDesc':
      list.sort((a, b) => (b.name || '').localeCompare(a.name || '', 'lo'));
      break;
    case 'memberCountDesc':
      list.sort((a, b) => ((b.members || []).length) - ((a.members || []).length));
      break;
    case 'memberCountAsc':
      list.sort((a, b) => ((a.members || []).length) - ((b.members || []).length));
      break;
    case 'unitAsc':
    default:
      list.sort((a, b) => naturalUnitCompare(a.unit || '', b.unit || ''));
      break;
  }

  return list;
}

          type: 'member',
          householdCode: household.householdCode,
          memberCode: member.memberCode
        }, color);
        allRows.push(memberRow);
      });
    }
  });
  
  totalRows = allRows.length;
  const totalPages = Math.ceil(totalRows / rowsPerPage);
  
  // Display rows for current page
  const startIndex = (currentPage - 1) * rowsPerPage;
  const endIndex = Math.min(startIndex + rowsPerPage, totalRows);
  
  for (let i = startIndex; i < endIndex; i++) {
    tbody.appendChild(allRows[i]);
  }
  
  // Update pagination UI
  updatePaginationUI(totalPages);
  
  console.log('>>> Displayed rows ' + (startIndex + 1) + '-' + endIndex + ' of ' + totalRows);
  hideLoading();
}

function updatePaginationUI(totalPages) {
  document.getElementById('currentPage').textContent = currentPage;
  document.getElementById('totalPages').textContent = totalPages;
  
  document.getElementById('btnFirst').disabled = currentPage === 1;
  document.getElementById('btnPrev').disabled = currentPage === 1;
  document.getElementById('btnNext').disabled = currentPage === totalPages;
  document.getElementById('btnLast').disabled = currentPage === totalPages;
  
  document.getElementById('pagination').style.display = totalPages > 1 ? 'flex' : 'none';
}

function changePage(direction) {
  const totalPages = Math.ceil(totalRows / rowsPerPage);
  
  switch(direction) {
    case 'first':
      currentPage = 1;
      break;
    case 'prev':
      if (currentPage > 1) currentPage--;
      break;
    case 'next':
      if (currentPage < totalPages) currentPage++;
      break;
    case 'last':
      currentPage = totalPages;
      break;
  }
  
  loadPopulationData();
  
  // Scroll to top of table
  document.getElementById('dataTable').scrollIntoView({ behavior: 'smooth', block: 'start' });
}

function performLocalHouseholdSearch(searchTerm) {
  if (!searchTerm) {
    currentPage = 1;
    loadPopulationData();
    return;
  }
  
  const searchLower = searchTerm.toLowerCase();
  const results = [];
  
  householdsData.forEach(household => {
    let match =
      (household.name && household.name.toLowerCase().includes(searchLower)) ||
      (household.phone && household.phone.includes(searchTerm)) ||
      (household.householdCode && household.householdCode.includes(searchTerm)) ||
      (household.houseNumber && household.houseNumber.toLowerCase().includes(searchLower));
    
    if (!match && household.members && household.members.length > 0) {
      match = household.members.some(m =>
        (m.name && m.name.toLowerCase().includes(searchLower)) ||
        (m.phone && m.phone.includes(searchTerm))
      );
    }
    
    if (match) {
      results.push(household);
    }
  });
  
  if (results.length === 0) {
    const tbody = document.getElementById('dataTableBody');
    tbody.innerHTML = `
      <tr>
        <td colspan="7" style="text-align: center; padding: 40px;">
          <div style="font-size: 48px; margin-bottom: 10px;">🔍</div>
          <div>ບໍ່ພົບຂໍ້ມູນທີ່ຄົ້ນຫາ: "${searchTerm}"</div>
          <button onclick="clearSearch()" class="btn btn-primary" style="margin-top: 15px;">
            ❌ ລ້າງການຄົ້ນຫາ
          </button>
        </td>
      </tr>
    `;
    document.getElementById('pagination').style.display = 'none';
    return;
  }
  
  displaySearchResults(results, searchTerm);
}

// ฟังก์ชันตรวจสอบฟอร์มสมาชิก
function validateMemberForm(memberDiv) {
  const errors = [];
  const requiredFields = memberDiv.querySelectorAll('.required-field');
  
  requiredFields.forEach(field => {
    const value = field.value.trim();
    const fieldName = field.getAttribute('data-field-name') || 'ຊ່ອງນີ້';
    
    if (!value) {
      errors.push({
        field: field,
        fieldName: fieldName,
        message: `${fieldName} ຍັງບໍ່ໄດ້ຕື່ມຂໍ້ມູນ`
      });
      
      // เพิ่มคลาส invalid
      field.classList.add('invalid');
    } else {
      // ลบคลาส invalid ถ้ามีค่า
      field.classList.remove('invalid');
    }
  });
  
  return errors;
}

// ฟังก์ชันเรียงลำดับสมาชิกใหม่
function reorderMembers() {
  const container = document.getElementById('membersContainer');
  const memberDivs = container.querySelectorAll('.member-section');
  
  memberDivs.forEach((div, index) => {
    const newNumber = index + 1;
    
    // อัปเดตหัวข้อ
    const heading = div.querySelector('h3');
    if (heading) {
      heading.setAttribute('data-member-number', newNumber); // เพิ่มบรรทัดนี้
      const span = heading.querySelector('span');
      if (span) {
        span.textContent = `👤 ສະມາຊິກຄອບຄົວ ຄົນທີ່ ${newNumber}`;
      }
    }
    
    // อัปเดต ID ของ div
    const oldId = div.id.match(/\d+/)[0];
    div.id = `member-${newNumber}`;
    
    // อัปเດต ID ของ inputs และ buttons ภายใน
    updateElementIds(div, oldId, newNumber);
  });
  
  // อัปเดต memberCounter
  memberCounter = memberDivs.length;
  
  // อัพเดต relationship options (เพิ่มใหม่)
  updateRelationshipOptions();
}

// ฟังก์ชันช่วยอัปเดต IDs

  const photoInput = container.querySelector(`#memberPhoto${oldId}`);

    const photoUpload = container.querySelector('.photo-upload');
    if (photoUpload) {
      photoUpload.setAttribute('onclick', `document.getElementById('memberPhoto${newId}').click()`);
    }
  }
  
  // อัปเดต Photo Preview
  const photoPreview = container.querySelector(`#memberPhotoPreview${oldId}`);

    photoPreview.id = `memberPhotoPreview${newId}`;
    if (photoInput) {
      photoInput.setAttribute('onchange', `previewPhoto(this, 'memberPhotoPreview${newId}')`);

}

function validateAllFormFields() {
  const errors = [];
  
  // ตรวจหัวหน้าครอบครัว
  const householdName = document.getElementById('householdName');
  if (!householdName.value.trim()) {
    errors.push({ field: 'householdName', message: 'ຊື່ຫົວໜ້າຄອບຄົວ ບໍ່ກົນຣຸນາກັນກໍ່' });
  }
  
  // ตรวจสมาชิก
  const memberForms = document.querySelectorAll('#membersContainer .member-section');
  memberForms.forEach((form, index) => {
    const memberErrors = validateMemberForm(form);
    errors.push(...memberErrors);
  });
  
  // ❌ ถ้ามี error
  if (errors.length > 0) {
    // Highlight ช่องที่ผิด
    errors.forEach(error => {
      const field = document.getElementById(error.field);
      if (field) {
        field.classList.add('invalid');
      }
    });
    
    // แสดง Popup
    showValidationPopup(errors, 0);
    return false;
  }
  
  return true;
}

function handleSaveSuccess(result) {
    console.log('Save success:', result);
    
    if (result.success) {

      const finalHouseholdData = result.household || householdData;
      
      if (result.householdCode) {
        finalHouseholdData.householdCode = result.householdCode;
      }
      
      const existingIndex = householdsData.findIndex(h => h.householdCode === finalHouseholdData.householdCode);
      if (existingIndex >= 0) {
        householdsData[existingIndex] = finalHouseholdData;
      } else {
        householdsData.push(finalHouseholdData);
      }
      
      saveToIndexedDB('households', finalHouseholdData);
      
      hideLoading();
      showToast(result.message, 'success');
      closeHouseholdForm();
      loadPopulationData();
    } else {
      hideLoading();
      showToast(result.message, 'error');
    }
  }
