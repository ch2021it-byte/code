function switchAuthTab(tab) {
  document.getElementById('form-login').classList.toggle('active', tab==='login');
  document.getElementById('form-register').classList.toggle('active', tab==='register');
  document.getElementById('tab-login').classList.toggle('active', tab==='login');
  document.getElementById('tab-register').classList.toggle('active', tab==='register');
}

function doLogin() {
  var u = document.getElementById('login-user').value.trim();
  var p = document.getElementById('login-pass').value.trim();
  var errEl = document.getElementById('login-err');
  var btn   = document.getElementById('btn-login');
  errEl.style.display = 'none';
  if (!u || !p) { errEl.textContent = 'กรุณากรอกชื่อผู้ใช้และรหัสผ่าน'; errEl.style.display = 'block'; return; }
  btn.disabled = true; btn.textContent = '⏳ กำลังตรวจสอบ...';
  google.script.run
    .withSuccessHandler(function(res) {
      btn.disabled = false; btn.textContent = 'เข้าสู่ระบบ →';
      if (res && res.success) {
        sessionStorage.setItem('eduUser', JSON.stringify(res.user));
        showAdminSection();
      } else {
        errEl.textContent = (res && res.error) || 'เข้าสู่ระบบไม่สำเร็จ';
        errEl.style.display = 'block';
      }
    })
    .withFailureHandler(function(err) {
      btn.disabled = false; btn.textContent = 'เข้าสู่ระบบ →';
      errEl.textContent = err.message || 'เกิดข้อผิดพลาด'; errEl.style.display = 'block';
    })
    .loginUser(u, p);
}

function doRegister() {
  var name = document.getElementById('reg-name').value.trim();
  var u    = document.getElementById('reg-user').value.trim();
  var p    = document.getElementById('reg-pass').value.trim();
  var errEl = document.getElementById('reg-err');
  var btn   = document.getElementById('btn-register');
  errEl.style.display = 'none';
  if (!name || !u || !p) { errEl.textContent = 'กรุณากรอกข้อมูลให้ครบ'; errEl.style.display = 'block'; return; }
  btn.disabled = true; btn.textContent = '⏳ กำลังลงทะเบียน...';
  google.script.run
    .withSuccessHandler(function(res) {
      btn.disabled = false; btn.textContent = 'ลงทะเบียน →';
      if (res && res.success) {
        document.getElementById('reg-fields').style.display = 'none';
        document.getElementById('reg-wait').style.display = 'block';
      } else {
        errEl.textContent = (res && res.error) || 'ลงทะเบียนไม่สำเร็จ'; errEl.style.display = 'block';
      }
    })
    .withFailureHandler(function(err) {
      btn.disabled = false; btn.textContent = 'ลงทะเบียน →';
      errEl.textContent = err.message || 'เกิดข้อผิดพลาด'; errEl.style.display = 'block';
    })
    .registerUser(u, p, name);
}

var _draftKey = 'edu_draft_' + ((_currentUser && _currentUser.username) || 'anon');

function draftSave() {
  clearTimeout(_draftTimer);
  _draftTimer = setTimeout(function() {
    var title = document.getElementById('quiz-title').value;
    var draft = { title: title, questions: questions, seq: qIdSeq, ts: Date.now() };
    try { localStorage.setItem(_draftKey, JSON.stringify(draft)); } catch(e) {}
    var ind = document.getElementById('draft-indicator');
    if (ind) { ind.style.display = 'inline-flex'; setTimeout(function(){ ind.style.display='none'; }, 2000); }
  }, 1200);

function checkDraftExists() {
  try {
    var raw = localStorage.getItem(_draftKey);
    if (!raw) return;
    var d = JSON.parse(raw);
    if (d && d.questions && d.questions.length > 0) {
      document.getElementById('btn-restore-draft').style.display = 'inline-flex';
    }
  } catch(e) {}
}

function restoreDraft() {
  try {
    var raw = localStorage.getItem(_draftKey);
    if (!raw) { toast('ไม่พบ Draft', 'error'); return; }
    var d = JSON.parse(raw);
    if (!confirm('กู้คืน Draft: "' + (d.title||'ไม่มีชื่อ') + '" จำนวน ' + (d.questions||[]).length + ' ข้อ?')) return;
    questions = (d.questions || []).map(function(q,i) { return Object.assign({},q,{id:++qIdSeq}); });
    qIdSeq = d.seq || questions.length;
    document.getElementById('quiz-title').value = d.title || '';
    renderQuestions();
    document.getElementById('btn-restore-draft').style.display = 'none';
    toast('กู้คืน Draft แล้ว ✅','success');
  } catch(e) { toast('กู้ Draft ไม่ได้: '+e.message,'error'); }
}

function clearDraft() {
  try { localStorage.removeItem(_draftKey); } catch(e) {}
  document.getElementById('btn-restore-draft').style.display = 'none';
}

function renderQBody(q){
  var body=document.getElementById('q-body-'+q.id); if(!body)return;
  var hint=(Q_TYPES.find(function(t){return t.val===q.type;})||{}).hint||'';
  if(q.type==='mc'||q.type==='dropdown'){
    var h='<div class="choices-wrap">';
    q.choices.forEach(function(ch,ci){
      var L=String.fromCharCode(65+ci), isC=q.correctAnswer===ci;
      h+='<div class="choice-row">'+
        '<div class="choice-mark'+(isC?' correct':'')+'" onclick="setCorrectMC('+q.id+','+ci+')">'+( isC?'✓':L)+'</div>'+
        '<input type="text" placeholder="ตัวเลือก '+L+'..." value="'+esc(ch)+'" oninput="updateChoice('+q.id+','+ci+',this.value);draftSave()">'+
        (q.choices.length>2?'<button class="choice-del-btn" onclick="removeChoice('+q.id+','+ci+')">✕</button>':'')+
        (isC?'<span class="correct-badge">✔ เฉลย</span>':'')+
      '</div>';
    });
    h+='</div><div style="margin-top:8px;display:flex;gap:8px;align-items:center;flex-wrap:wrap;">'+
      (q.choices.length<6?'<button class="btn btn-outline btn-sm" onclick="addChoice('+q.id+')">➕ ตัวเลือก</button>':'')+
      '<span style="font-size:11px;color:var(--text3);">เฉลย: <strong style="color:var(--teal)">ข้อ '+(q.correctAnswer+1)+'</strong></span>'+
    '</div><div class="type-hint">💡 '+hint+'</div>';
    body.innerHTML=h;
  } else if(q.type==='checkbox'){
    var ca=Array.isArray(q.correctAnswer)?q.correctAnswer:[];
    var h='<div class="choices-wrap">';
    q.choices.forEach(function(ch,ci){
      var L=String.fromCharCode(65+ci), isC=ca.includes(ci);
      h+='<div class="choice-row">'+
        '<div class="choice-mark chk'+(isC?' correct':'')+'" onclick="toggleCorrectChk('+q.id+','+ci+')">'+( isC?'✓':L)+'</div>'+
        '<input type="text" placeholder="ตัวเลือก '+L+'..." value="'+esc(ch)+'" oninput="updateChoice('+q.id+','+ci+',this.value);draftSave()">'+
        (q.choices.length>2?'<button class="choice-del-btn" onclick="removeChoice('+q.id+','+ci+')">✕</button>':'')+
        (isC?'<span class="correct-badge">✔ เฉลย</span>':'')+
      '</div>';
    });
    var cl=ca.map(function(i){return String.fromCharCode(65+i);}).join(', ')||'ยังไม่ได้เลือก';
    h+='</div><div style="margin-top:8px;display:flex;gap:8px;align-items:center;flex-wrap:wrap;">'+
      (q.choices.length<6?'<button class="btn btn-outline btn-sm" onclick="addChoice('+q.id+')">➕ ตัวเลือก</button>':'')+
      '<span style="font-size:11px;color:var(--text3);">เฉลย: <strong style="color:var(--teal)">'+cl+'</strong></span>'+
    '</div><div class="type-hint">💡 '+hint+' — คลิกวงกลมเพื่อเลือกหลายเฉลย</div>';
    body.innerHTML=h;
  } else {
    body.innerHTML='<div class="type-hint" style="margin-top:8px;">✏️ '+hint+'</div>';
  }
}

function updateQ(id,f,v){var q=questions.find(function(q){return q.id===id;});if(q)q[f]=v;}
function changeType(id,type){
  var q=questions.find(function(q){return q.id===id;});if(!q)return; q.type=type;
  if(type==='mc'||type==='dropdown'){if(!q.choices||!q.choices.length)q.choices=['','','',''];if(typeof q.correctAnswer!=='number')q.correctAnswer=0;}
  else if(type==='checkbox'){if(!q.choices||!q.choices.length)q.choices=['','','',''];if(!Array.isArray(q.correctAnswer))q.correctAnswer=[];}
  else q.correctAnswer=null;
  renderQBody(q); draftSave();
}
function updateChoice(qid,ci,v){var q=questions.find(function(q){return q.id===qid;});if(q)q.choices[ci]=v;}
function setCorrectMC(qid,ci){var q=questions.find(function(q){return q.id===qid;});if(!q)return;q.correctAnswer=ci;renderQBody(q);draftSave();}
function toggleCorrectChk(qid,ci){
  var q=questions.find(function(q){return q.id===qid;});if(!q)return;
  if(!Array.isArray(q.correctAnswer))q.correctAnswer=[];
  var idx=q.correctAnswer.indexOf(ci);
  if(idx>-1)q.correctAnswer.splice(idx,1);else q.correctAnswer.push(ci);
  q.correctAnswer.sort(function(a,b){return a-b;});renderQBody(q);draftSave();
}
function addChoice(qid){var q=questions.find(function(q){return q.id===qid;});if(!q||q.choices.length>=6)return;q.choices.push('');renderQBody(q);draftSave();}
function removeChoice(qid,ci){
  var q=questions.find(function(q){return q.id===qid;});if(!q||q.choices.length<=2)return;
  q.choices.splice(ci,1);
  if(q.type==='mc'&&q.correctAnswer>=q.choices.length)q.correctAnswer=0;
  if(q.type==='checkbox')q.correctAnswer=(q.correctAnswer||[]).filter(function(i){return i<q.choices.length;});
  renderQBody(q);draftSave();
}

function saveQuiz() {
  if(!questions.length){toast('กรุณาเพิ่มคำถามอย่างน้อย 1 ข้อ','error');return;}
  var title=document.getElementById('quiz-title').value.trim();
  if(!title){toast('กรุณากรอกชื่อแบบทดสอบ','error');return;}
  for(var i=0;i<questions.length;i++){
    if(!questions[i].questionText.trim()){toast('กรุณากรอกคำถามทุกข้อ','error');return;}
    if(questions[i].type!=='text')for(var j=0;j<(questions[i].choices||[]).length;j++)
      if(!questions[i].choices[j].trim()){toast('กรุณากรอกตัวเลือกให้ครบ','error');return;}
  }
  var btn=document.getElementById('save-btn');btn.disabled=true;btn.onclick=null;btn.textContent='⏳ กำลังสร้าง Sheet...';
  var clean=questions.map(function(q){return{type:q.type||'mc',questionText:String(q.questionText||''),choices:(q.choices||[]).map(function(c){return String(c||'');}),correctAnswer:q.correctAnswer!==undefined?q.correctAnswer:null};});
  var payload=JSON.stringify({title:title,questions:clean,createdBy:(_currentUser&&_currentUser.username)||''});
  google.script.run
    .withSuccessHandler(function(res){
      btn.disabled=false; btn.textContent='💾 บันทึกและสร้างลิงก์';
      if(!res){toast('ไม่ได้รับข้อมูลกลับ','error');return;}
      if(res.success){
        _lastSavedCount=questions.length; _lastSavedTitle=title;
        var _su=res.shareUrl||'', _tid=res.templateId||'';
        clearDraft();
        if(_coverBase64&&_coverMime){
          btn.textContent='⏳ กำลังอัพโหลดรูป...';
          google.script.run.withSuccessHandler(function(ir){if(ir&&ir.success)_coverFileUrl=ir.fileUrl;showSharePopup(_su);}).withFailureHandler(function(){showSharePopup(_su);}).uploadCoverImage(_coverBase64,_coverMime,_tid);
        } else showSharePopup(_su);
      } else toast('เกิดข้อผิดพลาด: '+(res.error||''),'error');
    })
    .withFailureHandler(function(err){btn.disabled=false;btn.textContent='💾 บันทึกและสร้างลิงก์';btn.onclick=saveQuiz;toast((err&&err.message)||'เกิดข้อผิดพลาด','error');})
    .saveQuestions(payload);
}

function filterTemplates(){
  var q=(document.getElementById('tpl-search').value||'').toLowerCase();
  var now=Date.now();
  var list=_allTemplates.filter(function(t){
    if(q&&!t.title.toLowerCase().includes(q))return false;
    if(_filterActive==='today'){var d=new Date(t.createdAt);return(now-d.getTime())<86400000;}
    if(_filterActive==='week'){var d=new Date(t.createdAt);return(now-d.getTime())<7*86400000;}
    return true;
  });
  renderTemplatesList(list);
  document.getElementById('tpl-count-label').textContent=list.length+'/'+_allTemplates.length+' รายการ';
}

function exportScoresExcel(tid, title) {
  if(!tid){toast('กรุณาเลือกเทมเพลตก่อน','error');return;}
  toast('กำลังโหลดข้อมูลคะแนน...','');
  google.script.run.withSuccessHandler(function(res){
    if(!res||!res.success){toast('เกิดข้อผิดพลาด: '+(res?res.error:''),'error');return;}
    _generateExcel(res.title||title, res.rows||[]);
  }).getScoresForExport(tid);
}

function _generateExcel(title, rows) {
  // สร้าง CSV (UTF-8 BOM สำหรับ Excel)
  var bom='\uFEFF';
  var cols=['ลำดับ','ชื่อ-นามสกุล','เบอร์โทร','ประเภท','คะแนน','คะแนนเต็ม','เปอร์เซ็นต์','วันที่ทำ'];
  var lines=[cols.join(',')];
  rows.forEach(function(r,i){
    var d=r.timestamp?new Date(r.timestamp).toLocaleString('th-TH'):'';
    var type=r.testType==='pre'?'Pre-test':'Post-test';
    lines.push([i+1,'"'+String(r.name||'').replace(/"/g,'""')+'"','"'+String(r.phone||'')+'"',type,r.score,r.total,r.percent+'%','"'+d+'"'].join(','));
  });
  var csv=bom+lines.join('\n');
  var blob=new Blob([csv],{type:'text/csv;charset=utf-8;'});
  var url=URL.createObjectURL(blob);
  var a=document.createElement('a'); a.href=url;
  var safe=(title||'scores').replace(/[^\w\s\u0E00-\u0E7F_-]/g,'').trim()||'scores';
  a.download=safe+'_scores.csv';
  document.body.appendChild(a); a.click(); document.body.removeChild(a); URL.revokeObjectURL(url);
  toast('Export '+rows.length+' แถว ✅','success');
}

function _generateWord(quizTitle,quizQuestions){
