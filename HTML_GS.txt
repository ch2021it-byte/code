const DRIVE_FOLDER_ID = '1tuOflctVSw8n6aUgLnEme0frjN5gTs0D';

// ★ ขั้นตอนที่ 1: ใส่ Spreadsheet ID ของ Master Sheet ตรงนี้ ★
// วิธีหา ID: เปิด Google Sheet → ดู URL
// https://docs.google.com/spreadsheets/d/ [ID อยู่ตรงนี้] /edit
// ถ้าไม่ใส่ → ระบบจะสร้าง Master Sheet ใหม่อัตโนมัติในครั้งแรก
const MASTER_SS_ID = 'PASTE_YOUR_SPREADSHEET_ID_HERE';

const MASTER_SHEET_NAMES = {
  CONFIG:    'Config',
  TEMPLATES: 'Templates'
};

// ── doGet (แก้: ป้องกัน e หรือ e.parameter เป็น null) ──────────
function doGet(e) {
  const params     = (e && e.parameter) ? e.parameter : {};
  const page       = params.page       || 'admin';
  const token      = params.token      || '';
  const templateId = params.templateId || '';

  const tmpl = HtmlService.createTemplateFromFile('index');
  tmpl.page        = page;
  tmpl.token       = token;
  tmpl.templateId  = templateId;
  tmpl.deployUrl   = ScriptApp.getService().getUrl();
  tmpl.quizTitle   = templateId ? getTemplateTitleById(templateId) : 'แบบทดสอบ';

  return tmpl.evaluate()
    .setTitle(page === 'student' ? tmpl.quizTitle : 'ระบบสร้างแบบทดสอบ')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

// ── Master Spreadsheet ─────────────────────────────────────────
// แก้ไขหลัก: ไม่ใช้ getActiveSpreadsheet() อีกต่อไป
// เพราะ Web App Standalone ไม่มี Active Spreadsheet
function getMasterSS() {
  // วิธี 1: ใช้ ID ที่กำหนดไว้ด้านบนโดยตรง
  if (MASTER_SS_ID && MASTER_SS_ID !== 'PASTE_YOUR_SPREADSHEET_ID_HERE') {
    return SpreadsheetApp.openById(MASTER_SS_ID);
  }

  // วิธี 2: ดึง ID จาก Script Properties (กรณีเคยสร้างไว้แล้ว)
  const props   = PropertiesService.getScriptProperties();
  const savedId = props.getProperty('MASTER_SS_ID');
  if (savedId) {
    return SpreadsheetApp.openById(savedId);
  }

  // วิธี 3: สร้าง Master Sheet ใหม่อัตโนมัติ (ครั้งแรกเท่านั้น)
  const newSS = SpreadsheetApp.create('Pre-Post Test — Master Sheet');
  props.setProperty('MASTER_SS_ID', newSS.getId());

  try {
    const folder  = DriveApp.getFolderById(DRIVE_FOLDER_ID);
    const newFile = DriveApp.getFileById(newSS.getId());
    folder.addFile(newFile);
    DriveApp.getRootFolder().removeFile(newFile);
  } catch(folderErr) {
    Logger.log('folder move error (master): ' + folderErr.message);
  }

  Logger.log('★ สร้าง Master Sheet ใหม่ ID: ' + newSS.getId());
  Logger.log('★ กรุณาก็อป ID นี้ไปใส่ใน MASTER_SS_ID ด้านบนด้วย');
  return newSS;
}

function getMasterSheet(name) {
  const ss = getMasterSS();
  if (!ss) throw new Error('ไม่พบ Master Spreadsheet กรุณาตั้งค่า MASTER_SS_ID ใน Code.gs');
  let sh = ss.getSheetByName(name);
  if (!sh) {
    sh = ss.insertSheet(name);
    if (name === MASTER_SHEET_NAMES.CONFIG) {
      sh.appendRow(['key', 'value']);
    } else if (name === MASTER_SHEET_NAMES.TEMPLATES) {
      sh.appendRow(['templateId', 'title', 'sheetId', 'sheetUrl', 'shareToken', 'shareUrl', 'createdAt', 'questionCount']);
    }
  }
  return sh;
}

// ── Config (master) ───────────────────────────────────────────
function getConfig(key) {
  const sh   = getMasterSheet(MASTER_SHEET_NAMES.CONFIG);
  const data = sh.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === key) return data[i][1];
  }
  return null;
}

function setConfig(key, value) {
  const sh   = getMasterSheet(MASTER_SHEET_NAMES.CONFIG);
  const data = sh.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === key) {
      sh.getRange(i + 1, 2).setValue(value);
      return;
    }
  }
  sh.appendRow([key, value]);
}

// ── Template registry ─────────────────────────────────────────
function registerTemplate(templateId, title, sheetId, sheetUrl, shareToken, shareUrl, questionCount) {
  const sh = getMasterSheet(MASTER_SHEET_NAMES.TEMPLATES);
  sh.appendRow([templateId, title, sheetId, sheetUrl, shareToken, shareUrl, new Date().toISOString(), questionCount]);
}

function getTemplateRow(templateId) {
  const sh   = getMasterSheet(MASTER_SHEET_NAMES.TEMPLATES);
  const data = sh.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === templateId) return { row: i + 1, data: data[i] };
  }
  return null;
}

function getTemplateTitleById(templateId) {
  try {
    const found = getTemplateRow(templateId);
    return found ? found.data[1] : 'แบบทดสอบ';
  } catch(e) {
    Logger.log('getTemplateTitleById error: ' + e.message);
    return 'แบบทดสอบ';
  }
}

// ── Get all templates ─────────────────────────────────────────
function getTemplates() {
  try {
    const sh   = getMasterSheet(MASTER_SHEET_NAMES.TEMPLATES);
    const data = sh.getDataRange().getValues();
    const templates = [];
    for (let i = 1; i < data.length; i++) {
      if (!data[i][0]) continue;
templates.push({
        templateId:    data[i][0],
        title:         data[i][1],
        sheetId:       data[i][2],
        sheetUrl:      data[i][3],
        shareToken:    data[i][4],
        shareUrl:      data[i][5],
        createdAt:     data[i][6],
        questionCount: data[i][7],
        coverImageId:  data[i][8] || '',
        coverImageUrl: data[i][9] || ''
      });
    }
    return { success: true, templates: templates.reverse() };
  } catch(e) {
    Logger.log('getTemplates error: ' + e.message);
    return { success: false, error: e.message };
  }
}

// ── Save questions → create new Sheet in Drive ─────────────────
function saveQuestions(payload) {
  try {
    const parsed    = JSON.parse(payload);
    const questions = parsed.questions;
    const quizTitle = (parsed.title        || 'แบบทดสอบ').trim();
    const tplName   = quizTitle;

    // 1. สร้าง Spreadsheet ใหม่
    const fileName = tplName + ' — ' + Utilities.formatDate(
      new Date(), Session.getScriptTimeZone(), 'yyyyMMdd_HHmm'
    );
    const newSS = SpreadsheetApp.create(fileName);
    SpreadsheetApp.flush();
    Utilities.sleep(1000);

    // 2. ย้ายไปยัง Drive folder
    try {
      const folder  = DriveApp.getFolderById(DRIVE_FOLDER_ID);
      const newFile = DriveApp.getFileById(newSS.getId());
      folder.addFile(newFile);
      DriveApp.getRootFolder().removeFile(newFile);
    } catch(folderErr) {
      Logger.log('folder move error: ' + folderErr.message);
    }

    // 3. เปิด Spreadsheet ใหม่ผ่าน ID (ป้องกัน reference หลุด)
    const ss     = SpreadsheetApp.openById(newSS.getId());
    const sheets = ss.getSheets();

    // 4. ตั้งชื่อ Sheet แรกเป็น Questions
    const qSheet = sheets[0];
    qSheet.setName('Questions');
    qSheet.clearContents();
    qSheet.appendRow(['id', 'type', 'questionText', 'choices', 'correctAnswer', 'createdAt']);
    questions.forEach(function(q, i) {
      qSheet.appendRow([
        i + 1,
        q.type         || 'mc',
        q.questionText || '',
        JSON.stringify(q.choices       || []),
        JSON.stringify(q.correctAnswer !== undefined ? q.correctAnswer : null),
        new Date().toISOString()
      ]);
    });
    SpreadsheetApp.flush();

    // 5. สร้าง Responses sheet
    let respSheet = ss.getSheetByName('Responses');
    if (!respSheet) respSheet = ss.insertSheet('Responses');
    respSheet.clearContents();
    respSheet.appendRow(['timestamp', 'name', 'phone', 'testType', 'answers', 'score', 'total', 'percent']);
    SpreadsheetApp.flush();

    // 6. สร้าง Info sheet
    let infoSheet = ss.getSheetByName('Info');
    if (!infoSheet) infoSheet = ss.insertSheet('Info');
    infoSheet.clearContents();

    // 7. สร้าง token และ URL
    const templateId = Utilities.getUuid();
    const shareToken = Utilities.getUuid();
    const deployUrl  = ScriptApp.getService().getUrl();
    const shareUrl   = deployUrl + '?page=student&templateId=' + templateId + '&token=' + shareToken;

    infoSheet.appendRow(['key',          'value']);
    infoSheet.appendRow(['title',        quizTitle]);
    infoSheet.appendRow(['templateName', tplName]);
    infoSheet.appendRow(['templateId',   templateId]);
    infoSheet.appendRow(['shareToken',   shareToken]);
    infoSheet.appendRow(['shareUrl',     shareUrl]);
    infoSheet.appendRow(['createdAt',    new Date().toISOString()]);
    SpreadsheetApp.flush();

    // 8. ลงทะเบียนใน master Templates sheet
    registerTemplate(
      templateId, quizTitle, ss.getId(),
      ss.getUrl(), shareToken, shareUrl, questions.length
    );

    return {
      success:    true,
      shareUrl:   shareUrl,
      sheetUrl:   ss.getUrl(),
      templateId: templateId
    };

  } catch(e) {
    Logger.log('saveQuestions error: ' + e.message);
    return { success: false, error: e.message };
  }
}

// ── Get questions for student (by templateId) ─────────────────
function getQuestionsForTemplate(templateId) {
  try {
    const found = getTemplateRow(templateId);
    if (!found) return { success: false, error: 'ไม่พบแบบทดสอบนี้' };

    const sheetId = found.data[2];
    const ss      = SpreadsheetApp.openById(sheetId);
    const qSheet  = ss.getSheetByName('Questions');
    const data    = qSheet.getDataRange().getValues();
    const questions = [];
    for (let i = 1; i < data.length; i++) {
      if (!data[i][0]) continue;
      questions.push({
        id:            data[i][0],
        type:          data[i][1],
        questionText:  data[i][2],
        choices:       JSON.parse(data[i][3] || '[]'),
        correctAnswer: JSON.parse(data[i][4] !== '' ? data[i][4] : 'null')
      });
    }
    const title         = found.data[1];
    const coverImageUrl = found.data[9] || '';
    return { success: true, questions: questions, title: title, coverImageUrl: coverImageUrl };
  } catch(e) {
    Logger.log('getQuestionsForTemplate error: ' + e.message);
    return { success: false, error: e.message };
  }
}

// ── Submit response ───────────────────────────────────────────
function submitResponse(payload) {
  try {
    const parsed     = JSON.parse(payload);
    const name       = parsed.name;
    const phone      = parsed.phone;
    const testType   = parsed.testType;
    const answers    = parsed.answers;
    const templateId = parsed.templateId;

    const found = getTemplateRow(templateId);
    if (!found) return { success: false, error: 'ไม่พบแบบทดสอบ' };

    const sheetId   = found.data[2];
    const ss        = SpreadsheetApp.openById(sheetId);
    const respSheet = ss.getSheetByName('Responses');
    const qSheet    = ss.getSheetByName('Questions');

    // ตรวจสอบซ้ำ
    const respData = respSheet.getDataRange().getValues();
    let preCount = 0, postCount = 0;
    for (let i = 1; i < respData.length; i++) {
      if (String(respData[i][2]) === String(phone)) {
        if (respData[i][3] === 'pre')  preCount++;
        if (respData[i][3] === 'post') postCount++;
      }
    }
    if (testType === 'pre'  && preCount  >= 1) return { success: false, error: 'คุณได้ทำ Pre-test แล้ว' };
    if (testType === 'post' && postCount >= 1) return { success: false, error: 'คุณได้ทำ Post-test แล้ว' };
    if (testType === 'post' && preCount  === 0) return { success: false, error: 'กรุณาทำ Pre-test ก่อน' };

    // คำนวณคะแนน
    const qData = qSheet.getDataRange().getValues();
    let score = 0, gradable = 0;
    for (let i = 1; i < qData.length; i++) {
      const type    = qData[i][1];
      const correct = JSON.parse(qData[i][4] !== '' ? qData[i][4] : 'null');
      const given   = answers[i - 1];
      if (type === 'text') continue;
      gradable++;
      if (type === 'mc' || type === 'dropdown') {
        if (String(given) === String(correct)) score++;
      } else if (type === 'checkbox') {
        const correctArr = Array.isArray(correct) ? correct.map(String).sort() : [];
        const givenArr   = Array.isArray(given)   ? given.map(String).sort()   : [];
        if (JSON.stringify(correctArr) === JSON.stringify(givenArr)) score++;
      }
    }
    const total   = gradable;
    const percent = total > 0 ? Math.round((score / total) * 100) : 0;

    respSheet.appendRow([
      new Date().toISOString(),
      name, phone, testType,
      JSON.stringify(answers),
      score, total, percent
    ]);

    return { success: true, score: score, total: total, percent: percent };
  } catch(e) {
    Logger.log('submitResponse error: ' + e.message);
    return { success: false, error: e.message };
  }
}

// ── Summary for specific template ─────────────────────────────
function getTemplateSummary(templateId) {
  try {
    const found = getTemplateRow(templateId);
    if (!found) return { success: false, error: 'ไม่พบเทมเพลต' };

    const sheetId   = found.data[2];
    const ss        = SpreadsheetApp.openById(sheetId);
    const respSheet = ss.getSheetByName('Responses');
    const data      = respSheet.getDataRange().getValues();

    const pre = [], post = [];
    for (let i = 1; i < data.length; i++) {
      if (!data[i][0]) continue;
      const row = {
        name:    data[i][1],
        phone:   String(data[i][2]),
        score:   parseInt(data[i][5])   || 0,
        total:   parseInt(data[i][6])   || 0,
        percent: parseFloat(data[i][7]) || 0
      };
      if (data[i][3] === 'pre')       pre.push(row);
      else if (data[i][3] === 'post') post.push(row);
    }

    const avg = function(arr) {
      return arr.length
        ? Math.round(arr.reduce(function(s, r) { return s + r.percent; }, 0) / arr.length)
        : 0;
    };

    return {
      success:  true,
      title:    found.data[1],
      sheetUrl: found.data[3],
      shareUrl: found.data[5],
      pre:  { count: pre.length,  avgPercent: avg(pre),  rows: pre  },
      post: { count: post.length, avgPercent: avg(post), rows: post }
    };
  } catch(e) {
    Logger.log('getTemplateSummary error: ' + e.message);
    return { success: false, error: e.message };
  }
}

// ── Delete template ───────────────────────────────────────────
function deleteTemplate(templateId) {
  try {
    const sh   = getMasterSheet(MASTER_SHEET_NAMES.TEMPLATES);
    const data = sh.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      if (data[i][0] === templateId) {
        const sheetId = data[i][2];
        // ลบ Google Sheet ใน Drive ด้วย
        try {
          DriveApp.getFileById(sheetId).setTrashed(true);
        } catch(driveErr) {
          Logger.log('drive delete error: ' + driveErr.message);
        }
        sh.deleteRow(i + 1);
        return { success: true };
      }
    }
    return { success: false, error: 'ไม่พบเทมเพลต' };
  } catch(e) {
    Logger.log('deleteTemplate error: ' + e.message);
    return { success: false, error: e.message };
  }
}
// ── Delete multiple templates ──────────────────────────────────
function deleteTemplates(templateIds) {
  try {
    const results = [];
    for (let i = 0; i < templateIds.length; i++) {
      const res = deleteTemplate(templateIds[i]);
      results.push({ id: templateIds[i], success: res.success });
    }
    return { success: true, results: results };
  } catch(e) {
    Logger.log('deleteTemplates error: ' + e.message);
    return { success: false, error: e.message };
  }
}

// ── Update questions in existing template ──────────────────────
function updateQuestions(payload) {
  try {
    const parsed     = JSON.parse(payload);
    const templateId = parsed.templateId;
    const questions  = parsed.questions;
    const quizTitle  = (parsed.title || 'แบบทดสอบ').trim();

    const found = getTemplateRow(templateId);
    if (!found) return { success: false, error: 'ไม่พบเทมเพลต' };

    const sheetId = found.data[2];
    const ss      = SpreadsheetApp.openById(sheetId);
    const qSheet  = ss.getSheetByName('Questions');

    // ล้างข้อมูลเก่าแล้วเขียนใหม่
    qSheet.clearContents();
    qSheet.appendRow(['id','type','questionText','choices','correctAnswer','createdAt']);
    questions.forEach(function(q, i) {
      qSheet.appendRow([
        i + 1,
        q.type         || 'mc',
        q.questionText || '',
        JSON.stringify(q.choices       || []),
        JSON.stringify(q.correctAnswer !== undefined ? q.correctAnswer : null),
        new Date().toISOString()
      ]);
    });

    // อัปเดตชื่อและจำนวนคำถามใน Info sheet
    const infoSheet = ss.getSheetByName('Info');
    if (infoSheet) {
      const infoData = infoSheet.getDataRange().getValues();
      for (let i = 1; i < infoData.length; i++) {
        if (infoData[i][0] === 'title') infoSheet.getRange(i+1,2).setValue(quizTitle);
      }
    }

    // อัปเดตใน Master Templates sheet
    const sh       = getMasterSheet(MASTER_SHEET_NAMES.TEMPLATES);
    const masterData = sh.getDataRange().getValues();
    for (let i = 1; i < masterData.length; i++) {
      if (masterData[i][0] === templateId) {
        sh.getRange(i+1, 2).setValue(quizTitle);        // title
        sh.getRange(i+1, 8).setValue(questions.length); // questionCount
        break;
      }
    }

    SpreadsheetApp.flush();
    return { success: true };
  } catch(e) {
    Logger.log('updateQuestions error: ' + e.message);
    return { success: false, error: e.message };
  }
}
// ── Upload cover image ─────────────────────────────────────────
function uploadCoverImage(base64Data, mimeType, templateId) {
  try {
    const folder  = DriveApp.getFolderById(DRIVE_FOLDER_ID);
    const blob    = Utilities.newBlob(
      Utilities.base64Decode(base64Data), mimeType, 'cover_' + templateId
    );
    const file    = folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    const fileId  = file.getId();
    const fileUrl = 'https://lh3.googleusercontent.com/d/' + fileId;

    // บันทึก fileId ลง Master sheet
    const sh   = getMasterSheet(MASTER_SHEET_NAMES.TEMPLATES);
    const data = sh.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      if (data[i][0] === templateId) {
        // column 9 (index 8) = coverImageId
        // ลบรูปเก่าก่อน
        const oldId = data[i][8];
        if (oldId) {
          try { DriveApp.getFileById(oldId).setTrashed(true); } catch(e) {}
        }
        sh.getRange(i + 1, 9).setValue(fileId);
        sh.getRange(i + 1, 10).setValue(fileUrl);
        break;
      }
    }
    return { success: true, fileId: fileId, fileUrl: fileUrl };
  } catch(e) {
    Logger.log('uploadCoverImage error: ' + e.message);
    return { success: false, error: e.message };
  }
}

}

function setupDrag(img) {
  img.onmousedown = function(e) {
    _dragging  = true;
    _dragStartX = e.clientX - _imgOffsetX;
    _dragStartY = e.clientY - _imgOffsetY;
    e.preventDefault();
  };
  document.onmousemove = function(e) {
    if (!_dragging) return;
    _imgOffsetX = e.clientX - _dragStartX;
    _imgOffsetY = e.clientY - _dragStartY;
    updateCropTransform();
  };
  document.onmouseup = function() { _dragging = false; };
  // Touch support
  img.ontouchstart = function(e) {
    _dragging   = true;
    _dragStartX = e.touches[0].clientX - _imgOffsetX;
    _dragStartY = e.touches[0].clientY - _imgOffsetY;
  };
  img.ontouchmove = function(e) {
    if (!_dragging) return;
    _imgOffsetX = e.touches[0].clientX - _dragStartX;
    _imgOffsetY = e.touches[0].clientY - _dragStartY;
    updateCropTransform();
    e.preventDefault();
  };
  img.ontouchend = function() { _dragging = false; };
}

/* ━━ QUESTION ━━ */
function addQuestion(data) {
  const id = ++qIdSeq;
  const q  = data || { id:id, type:'mc', questionText:'', choices:['','','',''], correctAnswer:0 };
  if (!data) { q.id = id; questions.push(q); }
  renderQuestions();
  setTimeout(function(){
    const el = document.getElementById('q-'+id);
    if (el) el.scrollIntoView({behavior:'smooth',block:'center'});
  }, 100);
}

function removeQuestion(id) {
  questions = questions.filter(function(q){ return q.id !== id; });
  renderQuestions();
}

function renderQuestions() {
  const area = document.getElementById('questions-area');
  area.innerHTML = '';
  questions.forEach(function(q, idx) {
    const div = document.createElement('div');
    div.className = 'q-block';
    div.id = 'q-'+q.id;
    const typeOptions = Q_TYPES.map(function(t){
      return '<option value="'+t.val+'"'+(q.type===t.val?' selected':'')+'>'+t.label+'</option>';
    }).join('');
    const currentType = Q_TYPES.find(function(t){ return t.val===q.type; });
    const typeTitle   = currentType ? currentType.fullLabel : '';
    div.innerHTML =
      '<button class="q-del-btn" onclick="removeQuestion('+q.id+')">✕</button>'+
      '<div class="q-top">'+
        '<div class="q-badge">'+(idx+1)+'</div>'+
        '<input class="q-top-input" type="text" placeholder="พิมพ์คำถาม..." value="'+esc(q.questionText)+'" oninput="updateQ('+q.id+',\'questionText\',this.value)">'+
        '<select class="q-type-select" onchange="changeType('+q.id+',this.value)" title="'+typeTitle+'">'+typeOptions+'</select>'+
      '</div>'+
      '<div id="q-body-'+q.id+'"></div>';
    area.appendChild(div);
    renderQBody(q);
  });
  document.getElementById('q-count').textContent =
    questions.length ? questions.length+' คำถาม' : 'ยังไม่มีคำถาม';
}

function renderQBody(q) {
  const body = document.getElementById('q-body-'+q.id);
  if (!body) return;
  const hint = (Q_TYPES.find(function(t){ return t.val===q.type; }) || {}).hint || '';

  if (q.type==='mc' || q.type==='dropdown') {
    let html = '<div class="choices-wrap" id="choices-'+q.id+'">';
    q.choices.forEach(function(ch, ci){
      const letter = String.fromCharCode(65+ci);
      const isCorr = q.correctAnswer===ci;
      html += '<div class="choice-row">'+
        '<div class="choice-mark'+(isCorr?' correct':'')+'" onclick="setCorrectMC('+q.id+','+ci+')" title="คลิกเพื่อตั้งเป็นเฉลย">'+(isCorr?'✓':letter)+'</div>'+
        '<input type="text" placeholder="ตัวเลือก '+letter+'..." value="'+esc(ch)+'" oninput="updateChoice('+q.id+','+ci+',this.value)">'+
        (q.choices.length>2?'<button class="choice-del-btn" onclick="removeChoice('+q.id+','+ci+')">✕</button>':'')+
        (isCorr?'<span class="correct-badge">✔ เฉลย</span>':'')+
      '</div>';
    });
    html += '</div>'+
      '<div style="margin-top:10px;display:flex;gap:8px;align-items:center;flex-wrap:wrap;">'+
        (q.choices.length<6?'<button class="btn btn-outline btn-sm" onclick="addChoice('+q.id+')">➕ เพิ่มตัวเลือก</button>':'')+
        '<span style="font-size:12px;color:var(--text3);">เฉลย: <strong style="color:var(--success)">ข้อ '+(q.correctAnswer+1)+'</strong></span>'+
      '</div>'+
      '<div class="type-hint">💡 '+hint+'</div>';
    body.innerHTML = html;

  } else if (q.type==='checkbox') {
    const corrArr = Array.isArray(q.correctAnswer) ? q.correctAnswer : [];
    let html = '<div class="choices-wrap" id="choices-'+q.id+'">';
    q.choices.forEach(function(ch, ci){
      const letter = String.fromCharCode(65+ci);
      const isCorr = corrArr.includes(ci);
      html += '<div class="choice-row">'+
        '<div class="choice-mark chk'+(isCorr?' correct':'')+'" onclick="toggleCorrectChk('+q.id+','+ci+')" title="คลิกเพื่อเลือก/ยกเลิกเฉลย">'+(isCorr?'✓':letter)+'</div>'+
        '<input type="text" placeholder="ตัวเลือก '+letter+'..." value="'+esc(ch)+'" oninput="updateChoice('+q.id+','+ci+',this.value)">'+
        (q.choices.length>2?'<button class="choice-del-btn" onclick="removeChoice('+q.id+','+ci+')">✕</button>':'')+
        (isCorr?'<span class="correct-badge">✔ เฉลย</span>':'')+
      '</div>';
    });
    const corrLabels = corrArr.map(function(i){ return String.fromCharCode(65+i); }).join(', ') || 'ยังไม่ได้เลือก';
    html += '</div>'+
      '<div style="margin-top:10px;display:flex;gap:8px;align-items:center;flex-wrap:wrap;">'+
        (q.choices.length<6?'<button class="btn btn-outline btn-sm" onclick="addChoice('+q.id+')">➕ เพิ่มตัวเลือก</button>':'')+
        '<span style="font-size:12px;color:var(--text3);">เฉลย: <strong style="color:var(--success)">'+corrLabels+'</strong></span>'+
      '</div>'+
      '<div class="type-hint">💡 '+hint+' — คลิกที่วงกลมเพื่อเลือกหลายเฉลย</div>';
    body.innerHTML = html;

  } else if (q.type==='text') {
    body.innerHTML = '<div class="type-hint" style="margin-top:8px;">✏️ '+hint+' — ไม่ต้องกำหนดเฉลย ตรวจโดยผู้สอน</div>';
  }
}

function updateQ(id, field, val){ const q=questions.find(function(q){return q.id===id;}); if(q) q[field]=val; }
function changeType(id, type){
  const q=questions.find(function(q){return q.id===id;}); if(!q) return;
  q.type=type;
  if(type==='mc'||type==='dropdown'){
    if(!Array.isArray(q.choices)||!q.choices.length) q.choices=['','','',''];
    if(typeof q.correctAnswer!=='number') q.correctAnswer=0;
  } else if(type==='checkbox'){
    if(!Array.isArray(q.choices)||!q.choices.length) q.choices=['','','',''];
    if(!Array.isArray(q.correctAnswer)) q.correctAnswer=[];
  } else if(type==='text'){
    q.correctAnswer=null;
  }
  renderQBody(q);
}
function updateChoice(qid,ci,val){ const q=questions.find(function(q){return q.id===qid;}); if(q) q.choices[ci]=val; }
function setCorrectMC(qid,ci){ const q=questions.find(function(q){return q.id===qid;}); if(!q) return; q.correctAnswer=ci; renderQBody(q); }
function toggleCorrectChk(qid,ci){
  const q=questions.find(function(q){return q.id===qid;}); if(!q) return;
  if(!Array.isArray(q.correctAnswer)) q.correctAnswer=[];
  const idx=q.correctAnswer.indexOf(ci);
  if(idx>-1) q.correctAnswer.splice(idx,1); else q.correctAnswer.push(ci);
  q.correctAnswer.sort(function(a,b){return a-b;});
  renderQBody(q);
}
function addChoice(qid){
  const q=questions.find(function(q){return q.id===qid;}); if(!q||q.choices.length>=6) return;
  q.choices.push(''); renderQBody(q);
}
function removeChoice(qid,ci){
  const q=questions.find(function(q){return q.id===qid;}); if(!q||q.choices.length<=2) return;
  q.choices.splice(ci,1);
  if(q.type==='mc'&&q.correctAnswer>=q.choices.length) q.correctAnswer=0;
  if(q.type==='checkbox') q.correctAnswer=(q.correctAnswer||[]).filter(function(i){return i<q.choices.length;});
  renderQBody(q);
}

/* ━━ SAVE QUIZ ━━ */
function saveQuiz() {
  if (!questions.length) { toast('กรุณาเพิ่มคำถามอย่างน้อย 1 ข้อ','error'); return; }
  const title = document.getElementById('quiz-title').value.trim();
  if (!title) { toast('กรุณากรอกชื่อแบบทดสอบ','error'); return; }
  for (var i=0; i<questions.length; i++) {
    var q = questions[i];

function _generatePDF(title, questions) {
  function escH(s) {
    return String(s||'').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');
  }

  var typeColor = {mc:'#1d4ed8', checkbox:'#6d28d9', dropdown:'#c2410c', text:'#475569'};
  var typeBg    = {mc:'#eff6ff', checkbox:'#f5f3ff', dropdown:'#fff7ed', text:'#f8fafc'};

  var qHTML = '';
  questions.forEach(function(q, qi) {
    var tColor = typeColor[q.type] || '#475569';
    var tBg    = typeBg[q.type]   || '#f8fafc';

function _generateWord(quizTitle, quizQuestions) {
  function xe(s) {

  function rpr(o) {

    if (o.bold)  s += '<w:b/><w:bCs/>';

    s += '<w:rFonts w:ascii="TH Sarabun New" w:hAnsi="TH Sarabun New" w:cs="TH Sarabun New"/>';
    s += '</w:rPr>';
    return s;
  }

  function wr(text, o) {
    return '<w:r>'+rpr(o)+'<w:t xml:space="preserve">'+xe(text)+'</w:t></w:r>';
  }

  function wp(runs, o) {
    o = o || {};
    var pp = '<w:pPr>';
    if (o.align)   pp += '<w:jc w:val="'+o.align+'"/>';
    if (o.indent)  pp += '<w:ind w:left="'+o.indent+'"/>';
    if (o.before !== undefined || o.after !== undefined)
      pp += '<w:spacing w:before="'+(o.before||0)+'" w:after="'+(o.after||0)+'"/>';
    if (o.fill)    pp += '<w:shd w:val="clear" w:color="auto" w:fill="'+o.fill+'"/>';
    if (o.divider) pp += '<w:pBdr><w:bottom w:val="single" w:sz="4" w:space="1" w:color="e2e8f0"/></w:pBdr>';
    pp += '</w:pPr>';

  }

function _buildZip(files) {
  function sb(str) {
    var b=[];
    for (var i=0;i<str.length;i++) {
      var c=str.charCodeAt(i);
      if (c<128) b.push(c);
      else if (c<2048) b.push((c>>6)|192,(c&63)|128);
      else b.push((c>>12)|224,((c>>6)&63)|128,(c&63)|128);
    }
    return b;
  }
  function crc(bytes) {
    var t=[];
    for (var i=0;i<256;i++){var c=i;for(var j=0;j<8;j++)c=(c&1)?(0xEDB88320^(c>>>1)):(c>>>1);t[i]=c;}
    var r=0xFFFFFFFF;
    for (var k=0;k<bytes.length;k++) r=t[(r^bytes[k])&0xFF]^(r>>>8);
    return (r^0xFFFFFFFF)>>>0;
  }
  function u16(n){return [n&0xFF,(n>>8)&0xFF];}
  function u32(n){return [n&0xFF,(n>>8)&0xFF,(n>>16)&0xFF,(n>>24)&0xFF];}

  var lh=[],cd=[],off=0;
  files.forEach(function(f){
    var nb=sb(f.name),db=sb(f.data),cr=crc(db),sz=db.length;
    var loc=[0x50,0x4B,0x03,0x04].concat(u16(20),u16(0),u16(0),u16(0),u16(0),u32(cr),u32(sz),u32(sz),u16(nb.length),u16(0)).concat(nb).concat(db);
    lh.push(loc);
    cd.push([0x50,0x4B,0x01,0x02].concat(u16(20),u16(20),u16(0),u16(0),u16(0),u16(0),u32(cr),u32(sz),u32(sz),u16(nb.length),u16(0),u16(0),u16(0),u16(0),u32(0),u32(off)).concat(nb));
    off+=loc.length;
  });
  var cdb=[].concat.apply([],cd),cds=cdb.length;
  var eocd=[0x50,0x4B,0x05,0x06].concat(u16(0),u16(0),u16(files.length),u16(files.length),u32(cds),u32(off),u16(0));
  var all=[].concat.apply([],[].concat.apply([],lh)).concat(cdb).concat(eocd);
  return new Uint8Array(all);
}

function _dlBlob(uint8arr, filename, mime) {
  var blob=new Blob([uint8arr],{type:mime});
  var url=URL.createObjectURL(blob);
  var a=window.document.createElement('a');
  a.href=url; a.download=filename;
  window.document.body.appendChild(a);
  a.click();
  window.document.body.removeChild(a);
  URL.revokeObjectURL(url);
}

          '<input type="checkbox" value="'+ci+'">'+
          '<span class="opt-letter">'+String.fromCharCode(65+ci)+'.</span>'+
          '<span>'+esc(ch)+'</span></label>';
      }).join('');
    } else if(q.type==='dropdown'){
      inner += '<select class="s-select" id="drop-'+q.id+'" onchange="pickDrop('+q.id+',this.value)">'+
        '<option value="">— เลือกคำตอบ —</option>'+
        q.choices.map(function(ch,ci){
          return '<option value="'+ci+'">'+String.fromCharCode(65+ci)+'. '+esc(ch)+'</option>';
        }).join('')+'</select>';
    } else if(q.type==='text'){
      inner += '<textarea class="s-textarea" id="txt-'+q.id+'" placeholder="พิมพ์คำตอบของคุณที่นี่..." oninput="pickTxt('+q.id+',this.value)" rows="4"></textarea>';
    }
    div.innerHTML = inner;
    body.appendChild(div);
  });
}

function pickMC(qid,ci){
  _answers[qid]=ci;
  const q=_questions.find(function(q){return q.id===qid;});
  if(q) q.choices.forEach(function(_,i){
    const el=document.getElementById('opt-'+qid+'-'+i);
    if(el) el.classList.toggle('sel', i===ci);
  });
  updateProg();
}
function pickChk(qid,ci){
  if(!_answers[qid]) _answers[qid]=[];
  const arr=_answers[qid];
  const idx=arr.indexOf(ci);
  if(idx>-1) arr.splice(idx,1); else arr.push(ci);
  const el=document.getElementById('chk-'+qid+'-'+ci);
  if(el) el.classList.toggle('sel', arr.includes(ci));
  updateProg();
}
function pickDrop(qid,val){ _answers[qid]=val===''?undefined:parseInt(val); updateProg(); }
function pickTxt(qid,val){  _answers[qid]=val; updateProg(); }
function updateProg(){
  const done=Object.keys(_answers).filter(function(k){
    const v=_answers[k]; return v!==undefined&&v!==null&&v!==''&&!(Array.isArray(v)&&!v.length);
  }).length;
  document.getElementById('q-prog').textContent=done+'/'+_questions.length+' ข้อ';
}

function doSubmit(){
  const name  = document.getElementById('s-name').value.trim();
  const phone = document.getElementById('s-phone').value.trim();
  const done  = Object.keys(_answers).filter(function(k){
    const v=_answers[k]; return v!==undefined&&v!==null&&v!==''&&!(Array.isArray(v)&&!v.length);
  }).length;
  if(done<_questions.length)
    if(!confirm('ตอบแล้ว '+done+'/'+_questions.length+' ข้อ\nต้องการส่งเลยไหม?')) return;

  const ansArr = _questions.map(function(q){ return _answers[q.id]!==undefined?_answers[q.id]:null; });
  const btn = document.getElementById('btn-submit');
  btn.disabled=true; btn.textContent='⏳ กำลังส่ง...';

  google.script.run
    .withSuccessHandler(function(res){
      btn.disabled=false; btn.textContent='📤 ส่งคำตอบ';
      if(res && res.success) showResult(res.score, res.total, res.percent);
      else toast((res&&res.error)||'เกิดข้อผิดพลาด','error');
    })
    .withFailureHandler(function(err){ btn.disabled=false; btn.textContent='📤 ส่งคำตอบ'; toast(err.message,'error'); })
    .submitResponse(JSON.stringify({name:name, phone:phone, testType:_type, answers:ansArr, templateId:_templateId}));
}

function showResult(score,total,percent){
  goStep('step-result');
  const icon  = percent>=80?'🎉':percent>=60?'👍':'📚';
  const color = percent>=80?'var(--success)':percent>=60?'var(--warning)':'var(--danger)';
  const typeTH= _type==='pre'?'Pre-test':'Post-test';
  const msg   = percent>=80?'ยอดเยี่ยม! คะแนนผ่านเกณฑ์ 80% 🏆':percent>=60?'ทำได้ดี! ยังมีส่วนที่พัฒนาได้ต่อไป':'ไม่ต้องท้อ ลองทบทวนบทเรียนแล้วกลับมาใหม่นะ 💪';
  document.getElementById('r-icon').textContent  = icon;
  document.getElementById('r-pct').innerHTML     = '<span style="color:'+color+'">'+percent+'%</span>';
  document.getElementById('r-score').textContent = score+' / '+total+' คะแนน ('+typeTH+')';
  document.getElementById('r-msg').textContent   = msg;
}
